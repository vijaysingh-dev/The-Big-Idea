# Command Pattern

## At a Glance

|||
|---|---|
|**Type**|Behavioral|
|**Intent**|Encapsulate a request as an object — parameterize clients, queue requests, support undo|
|**Frequency in Backend**|★★★★☆ — job queues, undo/redo, audit logs, transactional outbox|

---

## The Problem It Solves

You need to: decouple the sender of a request from its executor, store requests for later/retry, or support undo. Calling a method directly gives you none of these.

---

## Full Backend Example: Job Queue with Commands

```python
# command.py
from abc import ABC, abstractmethod
from dataclasses import dataclass, field
from typing import Optional, Any
from uuid import UUID, uuid4
from datetime import datetime
from enum import Enum

class CommandStatus(Enum):
    PENDING = "PENDING"
    RUNNING = "RUNNING"
    COMPLETED = "COMPLETED"
    FAILED = "FAILED"
    COMPENSATED = "COMPENSATED"

@dataclass
class CommandResult:
    success: bool
    output: Any = None
    error: Optional[str] = None

class Command(ABC):
    """
    Base command — encapsulates a unit of work.
    Supports execution, undo (compensation), and serialization.
    """
    command_id: UUID = field(default_factory=uuid4)
    created_at: datetime = field(default_factory=datetime.now)
    
    @abstractmethod
    def execute(self) -> CommandResult: pass
    
    def can_compensate(self) -> bool:
        return False
    
    def compensate(self) -> CommandResult:
        raise NotImplementedError("This command does not support compensation")
    
    @abstractmethod
    def to_dict(self) -> dict: pass
    
    @classmethod
    @abstractmethod
    def from_dict(cls, data: dict) -> "Command": pass


# commands/send_email_command.py
@dataclass
class SendEmailCommand(Command):
    to: str
    subject: str
    body: str
    template_id: Optional[str] = None
    
    def execute(self) -> CommandResult:
        try:
            msg_id = email_service.send(
                to=self.to,
                subject=self.subject,
                body=self.body
            )
            return CommandResult(success=True, output={"message_id": msg_id})
        except Exception as e:
            return CommandResult(success=False, error=str(e))
    
    def to_dict(self):
        return {"type": "send_email", "to": self.to,
                "subject": self.subject, "body": self.body}
    
    @classmethod
    def from_dict(cls, data):
        return cls(to=data["to"], subject=data["subject"], body=data["body"])


# commands/charge_payment_command.py
@dataclass
class ChargePaymentCommand(Command):
    user_id: UUID
    amount_cents: int
    currency: str
    payment_method_id: str
    order_id: UUID
    _transaction_id: Optional[str] = field(default=None, repr=False)
    
    def execute(self) -> CommandResult:
        try:
            result = payment_gateway.charge(
                method_id=self.payment_method_id,
                amount=self.amount_cents,
                currency=self.currency,
                idempotency_key=str(self.command_id)
            )
            self._transaction_id = result.transaction_id
            return CommandResult(success=True, output={"transaction_id": result.transaction_id})
        except PaymentDeclinedError as e:
            return CommandResult(success=False, error=f"Payment declined: {e}")
    
    def can_compensate(self) -> bool:
        return self._transaction_id is not None
    
    def compensate(self) -> CommandResult:
        """Undo: refund the charge."""
        try:
            payment_gateway.refund(self._transaction_id, self.amount_cents)
            return CommandResult(success=True)
        except Exception as e:
            return CommandResult(success=False, error=str(e))
    
    def to_dict(self):
        return {"type": "charge_payment", "user_id": str(self.user_id),
                "amount_cents": self.amount_cents, "currency": self.currency,
                "payment_method_id": self.payment_method_id,
                "order_id": str(self.order_id)}
    
    @classmethod
    def from_dict(cls, data):
        return cls(user_id=UUID(data["user_id"]),
                  amount_cents=data["amount_cents"],
                  currency=data["currency"],
                  payment_method_id=data["payment_method_id"],
                  order_id=UUID(data["order_id"]))


# command_queue.py — Queue + Executor
import json
from threading import Thread
from queue import Queue as ThreadQueue

class CommandQueue:
    """
    In-process command queue with retry.
    Production: replace _queue with Redis/SQS/RabbitMQ.
    """
    
    def __init__(self, max_retries: int = 3, worker_count: int = 4):
        self._queue: ThreadQueue = ThreadQueue()
        self._history: list = []       # completed commands (for undo)
        self._max_retries = max_retries
        self._registry: dict = {}      # type string → Command class
        
        # Start worker threads
        for _ in range(worker_count):
            Thread(target=self._worker, daemon=True).start()
    
    def register(self, command_type: str, command_class):
        self._registry[command_type] = command_class
    
    def enqueue(self, command: Command, priority: int = 0):
        """Add command to queue for async execution."""
        self._queue.put((priority, command))
    
    def execute_now(self, command: Command) -> CommandResult:
        """Execute synchronously and return result."""
        return self._execute_with_retry(command)
    
    def undo_last(self) -> Optional[CommandResult]:
        """Compensate the most recently completed command."""
        if not self._history:
            return None
        
        last_command = self._history.pop()
        if last_command.can_compensate():
            return last_command.compensate()
        
        logger.warning(f"Command {last_command.command_id} does not support compensation")
        return None
    
    def _worker(self):
        while True:
            priority, command = self._queue.get()
            self._execute_with_retry(command)
            self._queue.task_done()
    
    def _execute_with_retry(self, command: Command) -> CommandResult:
        for attempt in range(self._max_retries + 1):
            result = command.execute()
            
            if result.success:
                self._history.append(command)
                return result
            
            if attempt < self._max_retries:
                wait = 2 ** attempt  # exponential backoff
                logger.warning(f"Command {command.command_id} failed (attempt {attempt+1}), "
                              f"retrying in {wait}s: {result.error}")
                time.sleep(wait)
            else:
                logger.error(f"Command {command.command_id} failed after {self._max_retries} retries")
        
        return result


# Usage
queue = CommandQueue()
queue.register("send_email", SendEmailCommand)
queue.register("charge_payment", ChargePaymentCommand)

# Enqueue for async execution
queue.enqueue(SendEmailCommand(
    to="user@example.com",
    subject="Welcome!",
    body="Thanks for signing up."
))

# Execute and undo
charge = ChargePaymentCommand(
    user_id=user_id,
    amount_cents=9999,
    currency="INR",
    payment_method_id="pm_123",
    order_id=order_id
)
result = queue.execute_now(charge)
if not result.success:
    queue.undo_last()  # refund if failed
```

## Real-World Backend Uses

|Command Use|Description|
|---|---|
|Background job queue|Serialize commands, process async|
|Saga steps|Each saga step is a command with compensation|
|Audit log|Store commands as immutable history|
|Undo/redo|Command history + compensate|
|Transactional outbox|Save command to DB, process after commit|
|Batch processing|Collect commands, execute in bulk|
|CLI operations|Each CLI action = a command|

## Transactional Outbox Pattern (Command + DB)

```python
# Save command to DB in same transaction as business data
# Worker picks up and executes — guarantees at-least-once delivery

class TransactionalOutbox:
    def store(self, command: Command, session):
        """Save command alongside business data in same transaction."""
        session.add(OutboxEntry(
            id=command.command_id,
            command_type=command.__class__.__name__,
            payload=json.dumps(command.to_dict()),
            created_at=datetime.now(),
            status="PENDING"
        ))
    
    def process_pending(self):
        """Worker: pick up and execute pending commands."""
        entries = outbox_repo.find_pending(limit=10)
        for entry in entries:
            command_class = self._registry[entry.command_type]
            command = command_class.from_dict(json.loads(entry.payload))
            result = command.execute()
            if result.success:
                entry.status = "DONE"
            else:
                entry.retry_count += 1
                if entry.retry_count >= MAX_RETRIES:
                    entry.status = "DEAD"
```

## Tradeoffs

|Pro|Con|
|---|---|
|Decouples sender from receiver|More classes per operation|
|Serializable — store, queue, retry|Command must be self-contained (no closure state)|
|Undo support built-in|Compensation logic complexity|
|Composable into macros|Command explosion for large systems|
