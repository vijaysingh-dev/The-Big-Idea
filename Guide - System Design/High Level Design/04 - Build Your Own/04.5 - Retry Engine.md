# Build Your Own: Retry Engine

> **Revision tags:** #build-your-own #resilience  
> **Related:** [[02.5 - Failure and Resilience]]

---

## Goal

Build a retry engine that handles transient failures with:
- Exponential backoff with jitter
- Configurable max attempts and timeout
- Retryable vs non-retryable error classification
- Async retry queue for long-running retries

---

## Core Design

Two modes:
1. **Inline retry** — retries happen synchronously in the calling thread (suitable for fast operations)
2. **Async retry queue** — failed operations are enqueued for background retry (suitable for long delays or decoupled retries)

## 1. Inline Retry with Exponential Backoff + Jitter

```python
import time
import random
import logging
from typing import TypeVar, Callable, Optional, Set, Type
from functools import wraps

logger = logging.getLogger(__name__)

T = TypeVar("T")

class RetryExhausted(Exception):
    """Raised when all retry attempts are exhausted."""
    def __init__(self, attempts: int, last_error: Exception):
        self.attempts = attempts
        self.last_error = last_error
        super().__init__(f"All {attempts} retry attempts failed. "
                         f"Last error: {last_error}")

class NonRetryableError(Exception):
    """Wrap an error to signal it should not be retried."""
    pass


def compute_backoff(attempt: int,
                    base_delay: float = 0.1,
                    max_delay: float = 30.0,
                    jitter: bool = True) -> float:
    """
    Exponential backoff: delay = min(cap, base * 2^attempt)
    Jitter randomizes within [0, delay] to spread retry storms.
    """
    delay = min(max_delay, base_delay * (2 ** attempt))
    if jitter:
        delay = random.uniform(0, delay)
    return delay


def retry(max_attempts: int = 3,
          base_delay: float = 0.1,
          max_delay: float = 30.0,
          retryable_exceptions: Optional[Set[Type[Exception]]] = None,
          on_retry: Optional[Callable] = None):
    """
    Decorator for retrying a function on failure.

    Args:
        max_attempts:         Maximum number of attempts (1 = no retry)
        base_delay:           Initial backoff delay in seconds
        max_delay:            Maximum delay cap in seconds
        retryable_exceptions: Set of exception types to retry on (None = all)
        on_retry:             Callback(attempt, error, delay) called before each retry
    """
    def decorator(func: Callable[..., T]) -> Callable[..., T]:
        @wraps(func)
        def wrapper(*args, **kwargs) -> T:
            last_error = None

            for attempt in range(max_attempts):
                try:
                    return func(*args, **kwargs)

                except NonRetryableError as e:
                    # Explicitly non-retryable — fail immediately
                    raise e.__cause__ or e

                except Exception as e:
                    last_error = e

                    # Check if this exception type is retryable
                    if retryable_exceptions and not isinstance(e, tuple(retryable_exceptions)):
                        logger.warning(f"Non-retryable error in {func.__name__}: {e}")
                        raise

                    is_last_attempt = (attempt == max_attempts - 1)
                    if is_last_attempt:
                        break

                    delay = compute_backoff(attempt, base_delay, max_delay)

                    logger.warning(
                        f"[Retry] {func.__name__} failed (attempt {attempt + 1}/{max_attempts}). "
                        f"Retrying in {delay:.2f}s. Error: {e}"
                    )

                    if on_retry:
                        on_retry(attempt + 1, e, delay)

                    time.sleep(delay)

            raise RetryExhausted(max_attempts, last_error)

        return wrapper
    return decorator


# ── Usage as decorator ────────────────────────────────────────────────────────

import requests as http_lib

@retry(max_attempts=4, base_delay=0.2, max_delay=10.0,
       retryable_exceptions={http_lib.Timeout, http_lib.ConnectionError})
def call_payment_service(order_id: int, amount: float) -> dict:
    response = http_lib.post(
        "http://payment-service/charge",
        json={"order_id": order_id, "amount": amount},
        timeout=2.0
    )
    if response.status_code == 400:
        # Bad request — retrying won't help
        raise NonRetryableError(f"Invalid payment data: {response.text}")
    response.raise_for_status()
    return response.json()
```

## 2. Retry as Context Manager (for inline use without decorator)

```python
from contextlib import contextmanager

@contextmanager
def retrying(max_attempts: int = 3, base_delay: float = 0.1, max_delay: float = 30.0):
    """
    Context manager for retry loops.

    Usage:
        with retrying(max_attempts=3) as attempt:
            result = do_something()
    """
    last_error = None
    for attempt in range(max_attempts):
        try:
            yield attempt
            return  # success — exit
        except Exception as e:
            last_error = e
            if attempt < max_attempts - 1:
                delay = compute_backoff(attempt, base_delay, max_delay)
                logger.warning(f"Attempt {attempt + 1} failed: {e}. Retrying in {delay:.2f}s")
                time.sleep(delay)
    raise RetryExhausted(max_attempts, last_error)
```

## 3. Async Retry Queue (for background retries with long delays)

```python
import threading
import heapq
from dataclasses import dataclass, field
from typing import Any

@dataclass(order=True)
class RetryTask:
    """A task scheduled for retry at a specific time."""
    run_at: float                        # unix timestamp
    attempt: int = field(compare=False)
    func: Callable = field(compare=False)
    args: tuple = field(compare=False)
    kwargs: dict = field(compare=False)
    max_attempts: int = field(compare=False)
    task_id: str = field(compare=False)


class AsyncRetryQueue:
    """
    Background retry queue. Failed operations are enqueued here
    and retried after their backoff delay expires.
    """

    def __init__(self, max_attempts: int = 5, base_delay: float = 1.0,
                 max_delay: float = 300.0):
        self.max_attempts = max_attempts
        self.base_delay = base_delay
        self.max_delay = max_delay

        self._heap: list[RetryTask] = []  # min-heap by run_at
        self._lock = threading.Lock()
        self._condition = threading.Condition(self._lock)
        self._thread = threading.Thread(target=self._run, daemon=True)
        self._thread.start()

    def enqueue(self, func: Callable, *args, **kwargs) -> str:
        """Schedule a function for immediate execution with retry on failure."""
        task_id = str(uuid.uuid4())
        task = RetryTask(
            run_at=time.time(),
            attempt=0,
            func=func,
            args=args,
            kwargs=kwargs,
            max_attempts=self.max_attempts,
            task_id=task_id
        )
        with self._condition:
            heapq.heappush(self._heap, task)
            self._condition.notify()
        return task_id

    def _schedule_retry(self, task: RetryTask):
        """Schedule a failed task for retry after backoff."""
        delay = compute_backoff(task.attempt, self.base_delay, self.max_delay)
        retry_task = RetryTask(
            run_at=time.time() + delay,
            attempt=task.attempt + 1,
            func=task.func,
            args=task.args,
            kwargs=task.kwargs,
            max_attempts=task.max_attempts,
            task_id=task.task_id
        )
        with self._condition:
            heapq.heappush(self._heap, retry_task)
            self._condition.notify()
        logger.info(f"[AsyncRetry] Task {task.task_id} scheduled for retry "
                    f"in {delay:.1f}s (attempt {task.attempt + 1}/{task.max_attempts})")

    def _run(self):
        """Background thread: execute tasks when their run_at time arrives."""
        while True:
            with self._condition:
                # Wait until there's a task ready
                while not self._heap or self._heap[0].run_at > time.time():
                    if self._heap:
                        wait_time = self._heap[0].run_at - time.time()
                        self._condition.wait(timeout=max(0, wait_time))
                    else:
                        self._condition.wait(timeout=5)

                if not self._heap:
                    continue
                task = heapq.heappop(self._heap)

            # Execute outside lock
            try:
                task.func(*task.args, **task.kwargs)
                logger.info(f"[AsyncRetry] Task {task.task_id} succeeded "
                             f"on attempt {task.attempt + 1}")
            except Exception as e:
                if task.attempt + 1 >= task.max_attempts:
                    logger.error(f"[AsyncRetry] Task {task.task_id} FAILED permanently "
                                 f"after {task.max_attempts} attempts. Error: {e}")
                    # Send to DLQ, alert on-call, etc.
                else:
                    self._schedule_retry(task)
```

## Usage

```python
# Async retry queue — fire and forget with background retries
retry_queue = AsyncRetryQueue(max_attempts=5, base_delay=1.0, max_delay=60.0)

def send_webhook(url: str, payload: dict):
    response = http_lib.post(url, json=payload, timeout=5)
    response.raise_for_status()

# Enqueue — returns immediately, retries happen in background
retry_queue.enqueue(send_webhook, "https://partner.com/webhook", {"event": "order.shipped"})
```

## What Production Systems Add

| Feature | What's Missing Here |
|---|---|
| Persistence | Tasks lost if process restarts; use Redis/DB for task state |
| Distributed workers | Multiple processes consuming from the same retry queue |
| Per-task retry config | Different delays for different operation types |
| Circuit breaker integration | Stop retrying when circuit is open |
| Idempotency keys | Ensure retried operations don't double-process |
| Metrics | Track retry rates, exhaustion rates per operation type |
| Dead letter handling | Alert on permanently failed tasks, enable manual replay |

---

## Key Takeaways

- Exponential backoff = delay doubles each attempt
- Jitter = randomize the delay to prevent thundering herd
- Classify errors before retrying — HTTP 400 should never be retried
- For long delays (> 10s), async queue is better than blocking a thread
- The heap (min-by-run_at) is the natural data structure for scheduled retries
