# Background Jobs and Schedulers

## When to Use Background Jobs

```
Use background jobs when:
  - Operation takes >200ms (PDF generation, email, export)
  - Operation can fail and needs retry (payment processing, external API calls)
  - Fan-out to many recipients (broadcast notifications)
  - Scheduled/recurring work (cleanup, reports, billing)
  - Non-critical post-processing (analytics, audit log)

Don't use background jobs when:
  - Caller needs the result immediately (synchronous response required)
  - Operation is fast (<50ms) and reliable
  - Ordering between operations must be strict
```

## Job Queue Architecture

```
┌─────────────────┐     enqueue     ┌──────────┐     dequeue    ┌─────────────────┐
│  Application    │ ─────────────▶  │  Queue   │ ──────────────▶│  Worker Pool    │
│  (Producer)     │                 │ (Redis/  │                 │                 │
│                 │                 │  SQS/RQ) │                 │  Worker 1       │
│ order_service   │                 └──────────┘                 │  Worker 2       │
│ .place_order()  │                      │                       │  Worker 3       │
│   ↓             │                      │ DLQ                   └─────────────────┘
│ queue.enqueue(  │                 ┌────▼─────┐                        │
│  SendEmail(..)) │                 │  Dead    │                        ▼
└─────────────────┘                 │  Letter  │               Job processing + retry
                                    │  Queue   │
                                    └──────────┘
```

## Full Implementation: Job Queue with Celery

```python
# celery_app.py
from celery import Celery
from celery.utils.log import get_task_logger

app = Celery("myapp",
             broker="redis://localhost:6379/0",
             backend="redis://localhost:6379/1")

app.conf.update(
    task_serializer="json",
    result_serializer="json",
    accept_content=["json"],
    task_acks_late=True,              # acknowledge AFTER processing (at-least-once)
    task_reject_on_worker_lost=True,  # re-queue if worker dies mid-task
    task_default_retry_delay=60,      # 1 minute between retries
    task_max_retries=3,
    worker_prefetch_multiplier=1,     # one task at a time per worker (fair dispatch)
)


# tasks.py
from celery_app import app
from celery.exceptions import Retry

logger = get_task_logger(__name__)

@app.task(
    bind=True,
    max_retries=3,
    default_retry_delay=60,
    queue="notifications",
    acks_late=True
)
def send_order_confirmation_email(self, order_id: str, user_id: str):
    """
    Send order confirmation email.
    Retries 3 times with exponential backoff on failure.
    """
    try:
        order = order_repo.get(order_id)
        user = user_repo.get(user_id)
        
        email_service.send_order_confirmation(
            to=user.email,
            order=order
        )
        
        logger.info(f"Order confirmation sent for {order_id}")
    
    except EmailServiceTemporaryError as e:
        # Transient error: retry with exponential backoff
        raise self.retry(
            exc=e,
            countdown=60 * (2 ** self.request.retries)  # 60s, 120s, 240s
        )
    
    except EmailServicePermanentError as e:
        # Permanent error: don't retry, log and alert
        logger.error(f"Permanent email failure for {order_id}: {e}")
        alert_ops(f"Email delivery failed permanently for order {order_id}")
        # Task ends without retry — goes to failure state


@app.task(bind=True, max_retries=5, queue="payments")
def process_subscription_renewal(self, subscription_id: str):
    """Monthly subscription charge — idempotent via idempotency key."""
    try:
        sub = subscription_repo.get(subscription_id)
        
        if sub.status != "ACTIVE":
            logger.info(f"Skipping renewal for {subscription_id} — status: {sub.status}")
            return
        
        # Idempotency key prevents double-charge on retry
        idempotency_key = f"renewal:{subscription_id}:{sub.current_period_end.isoformat()}"
        
        payment_service.charge(
            method_id=sub.payment_method_id,
            amount=sub.plan.price,
            idempotency_key=idempotency_key
        )
        
        sub.renew()
        subscription_repo.save(sub)
    
    except PaymentDeclinedError as e:
        # Card declined — notify user and suspend (don't retry)
        subscription_service.handle_payment_failure(subscription_id)
        return
    
    except Exception as e:
        raise self.retry(exc=e, countdown=300)  # retry in 5 min


# Enqueue from application code
def place_order(cmd):
    order = order_repo.save(Order.create(cmd))
    
    # Non-blocking: returns immediately, email sent asynchronously
    send_order_confirmation_email.delay(
        order_id=str(order.id),
        user_id=str(cmd.user_id)
    )
    
    return order
```

## Job Scheduler (Cron-like)

```python
# scheduler.py — Celery Beat for periodic tasks
from celery.schedules import crontab

app.conf.beat_schedule = {
    # Every day at 2 AM IST: send daily digest emails
    "daily-digest": {
        "task": "tasks.send_daily_digest",
        "schedule": crontab(hour=2, minute=0),
        "options": {"queue": "notifications"}
    },
    
    # Every hour: expire abandoned carts
    "expire-abandoned-carts": {
        "task": "tasks.expire_abandoned_carts",
        "schedule": crontab(minute=0),
    },
    
    # 1st of every month at midnight: process subscription renewals
    "subscription-renewals": {
        "task": "tasks.trigger_subscription_renewals",
        "schedule": crontab(day_of_month=1, hour=0, minute=0),
    },
    
    # Every 5 minutes: process transactional outbox
    "outbox-processor": {
        "task": "tasks.process_outbox",
        "schedule": 300.0,  # every 300 seconds
    }
}
```

## Distributed Job Scheduler Design

```python
# For cases where you need custom scheduling logic beyond cron

class DistributedJobScheduler:
    """
    Ensures a scheduled job runs exactly once across multiple nodes.
    Uses Redis distributed lock to prevent duplicate execution.
    """
    
    def __init__(self, redis_client, job_repo):
        self._redis = redis_client
        self._repo = job_repo
    
    def schedule(self, job_id: str, run_at: datetime, payload: dict):
        self._repo.save(ScheduledJob(
            id=job_id,
            run_at=run_at,
            payload=payload,
            status=JobStatus.SCHEDULED
        ))
    
    def claim_and_run(self, handler):
        """
        Called by scheduler workers periodically.
        Only one worker processes each due job.
        """
        due_jobs = self._repo.find_due(
            before=datetime.now(),
            status=JobStatus.SCHEDULED,
            limit=10
        )
        
        for job in due_jobs:
            lock_key = f"job:lock:{job.id}"
            
            # Try to acquire lock — only one worker wins
            acquired = self._redis.set(lock_key, "1", nx=True, ex=300)  # 5 min lock
            if not acquired:
                continue   # another worker is handling this job
            
            try:
                self._repo.mark_running(job.id)
                handler(job.payload)
                self._repo.mark_completed(job.id)
            
            except Exception as e:
                self._repo.mark_failed(job.id, error=str(e))
                
                if job.retry_count < job.max_retries:
                    # Re-schedule with backoff
                    retry_at = datetime.now() + timedelta(
                        seconds=60 * (2 ** job.retry_count)
                    )
                    self._repo.schedule_retry(job.id, retry_at)
            
            finally:
                self._redis.delete(lock_key)
```

## Job Monitoring

```python
# Always monitor:
# 1. Queue depth — growing queue = workers can't keep up
# 2. Job success/failure rate
# 3. Job processing latency (enqueue → complete time)
# 4. DLQ depth — unprocessable jobs piling up

class JobMetrics:
    def record_enqueued(self, queue: str, job_type: str):
        metrics.increment("job.enqueued", tags={"queue": queue, "type": job_type})
    
    def record_completed(self, queue: str, job_type: str, duration_ms: float):
        metrics.increment("job.completed", tags={"queue": queue, "type": job_type})
        metrics.histogram("job.duration_ms", duration_ms, 
                         tags={"queue": queue, "type": job_type})
    
    def record_failed(self, queue: str, job_type: str, final: bool = False):
        metrics.increment("job.failed", 
                         tags={"queue": queue, "type": job_type, "final": final})
```

## Retry Best Practices

```
Exponential backoff with jitter:
  Attempt 1: wait 1s  ± 0.1s jitter
  Attempt 2: wait 2s  ± 0.2s jitter
  Attempt 3: wait 4s  ± 0.4s jitter
  Attempt 4: wait 8s  ± 0.8s jitter
  Attempt 5: →DLQ

Jitter prevents synchronized retry storms from many workers.

Always retry: network timeouts, rate limit errors, temporary DB unavailability
Never retry:  validation errors, authentication errors, business logic errors
```
