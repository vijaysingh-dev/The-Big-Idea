# Machine Coding Drills

The implementations asked for in coding rounds. Write each from memory, then compare — and note the edge cases listed, since those are usually what the question is actually testing.

## Utilities

**`debounce` with cancel**

```js
function debounce(fn, delay) {
  let timer;

  const debounced = (...args) => {
    clearTimeout(timer);
    timer = setTimeout(() => fn(...args), delay);
  };

  debounced.cancel = () => clearTimeout(timer);
  debounced.flush = (...args) => { clearTimeout(timer); fn(...args); };

  return debounced;
}
```

Volunteer: `cancel` for component cleanup, and leading/trailing edge options. → [[29.1 - Debounce and Throttle]]

**`throttle` with a trailing edge**

```js
function throttle(fn, interval) {
  let last = 0, timer;

  return (...args) => {
    const remaining = interval - (Date.now() - last);

    if (remaining <= 0) {
      last = Date.now();
      fn(...args);
    } else if (!timer) {
      timer = setTimeout(() => {
        last = Date.now();
        timer = null;
        fn(...args);
      }, remaining);
    }
  };
}
```

The trailing edge is the detail most candidates omit — without it the final event is dropped.

**`memoize` with a custom key**

```js
function memoize(fn, keyFn = (...args) => JSON.stringify(args)) {
  const cache = new Map();

  return (...args) => {
    const key = keyFn(...args);
    if (!cache.has(key)) cache.set(key, fn(...args));
    return cache.get(key);
  };
}
```

Volunteer: unbounded growth needs an LRU cap, and a `WeakMap` keyed by object identity avoids retaining keys. → [[02.4 - Closure Patterns in Production]]

**`once`**

```js
function once(fn) {
  let called = false, result;

  return (...args) => {
    if (!called) { called = true; result = fn(...args); }
    return result;
  };
}
```

Setting `called` **before** invoking guards against re-entrant calls.

**`deepClone` with cycle handling**

```js
function deepClone(value, seen = new WeakMap()) {
  if (value === null || typeof value !== "object") return value;
  if (seen.has(value)) return seen.get(value);

  if (value instanceof Date) return new Date(value);
  if (value instanceof RegExp) return new RegExp(value.source, value.flags);

  const result = Array.isArray(value) ? [] : Object.create(Object.getPrototypeOf(value));
  seen.set(value, result);                          // BEFORE recursing

  if (value instanceof Map) {
    value.forEach((v, k) => result.set(deepClone(k, seen), deepClone(v, seen)));
    return result;
  }
  if (value instanceof Set) {
    value.forEach(v => result.add(deepClone(v, seen)));
    return result;
  }

  for (const key of Reflect.ownKeys(value)) result[key] = deepClone(value[key], seen);
  return result;
}
```

The `WeakMap` must be populated before recursing, or cycles infinite-loop. Volunteer `structuredClone` as the built-in answer and what it refuses to clone. → [[03.5 - Shallow vs Deep Copy]]

**`pipe` and `compose`**

```js
const pipe = (...fns) => (x) => fns.reduce((acc, fn) => fn(acc), x);
const compose = (...fns) => (x) => fns.reduceRight((acc, fn) => fn(acc), x);
```

**`curry`**

```js
function curry(fn) {
  return function curried(...args) {
    if (args.length >= fn.length) return fn.apply(this, args);
    return (...rest) => curried.apply(this, [...args, ...rest]);
  };
}
```

Volunteer: `fn.length` breaks with default and rest parameters. → [[02.9 - Currying and Partial Application]]

## Polyfills

**`Array.prototype.map`**

```js
Array.prototype.myMap = function (callback, thisArg) {
  if (typeof callback !== "function") throw new TypeError();

  const result = new Array(this.length);
  for (let i = 0; i < this.length; i++) {
    if (i in this) result[i] = callback.call(thisArg, this[i], i, this);
  }
  return result;
};
```

`if (i in this)` skips holes in sparse arrays — the detail that separates a real implementation. Same shape for `filter`, `forEach` and `reduce`.

**`Function.prototype.bind`**

```js
Function.prototype.myBind = function (context, ...preset) {
  const fn = this;

  function bound(...later) {
    const isNew = this instanceof bound;
    return fn.apply(isNew ? this : context, [...preset, ...later]);
  }

  bound.prototype = Object.create(fn.prototype ?? null);
  return bound;
};
```

Two things interviewers look for: `new` overrides the bound `this`, and the prototype is copied so `instanceof` works. → [[02.8 - call apply and bind]]

**`Promise.all`**

```js
function promiseAll(promises) {
  return new Promise((resolve, reject) => {
    const results = new Array(promises.length);
    let remaining = promises.length;

    if (remaining === 0) return resolve([]);

    promises.forEach((p, i) => {
      Promise.resolve(p).then(
        (value) => {
          results[i] = value;
          if (--remaining === 0) resolve(results);
        },
        reject
      );
    });
  });
}
```

Index-based writes preserve order; `Promise.resolve(p)` handles non-promise values; the empty-array case must resolve immediately. → [[04.5 - Promise Combinators]]

**A minimal Promise**

```js
class MyPromise {
  #state = "pending";
  #value;
  #callbacks = [];

  constructor(executor) {
    const settle = (state, value) => {
      if (this.#state !== "pending") return;          // one-shot
      this.#state = state;
      this.#value = value;
      this.#callbacks.forEach(cb => queueMicrotask(cb));
      this.#callbacks = [];
    };

    try {
      executor(v => settle("fulfilled", v), e => settle("rejected", e));
    } catch (e) {
      settle("rejected", e);
    }
  }

  then(onFulfilled, onRejected) {
    return new MyPromise((resolve, reject) => {
      const handle = () => {
        const handler = this.#state === "fulfilled" ? onFulfilled : onRejected;

        if (typeof handler !== "function") {
          this.#state === "fulfilled" ? resolve(this.#value) : reject(this.#value);
          return;
        }

        try { resolve(handler(this.#value)); } catch (e) { reject(e); }
      };

      this.#state === "pending" ? this.#callbacks.push(handle) : queueMicrotask(handle);
    });
  }

  catch(onRejected) { return this.then(undefined, onRejected); }
}
```

The pieces being tested: the one-shot state guard, `queueMicrotask` scheduling, `then` returning a new promise, errors converted to rejections, and pass-through when a handler is missing. → [[04.2 - Promises from First Principles]]

## Async patterns

**Retry with exponential backoff and jitter**

```js
async function retry(fn, { attempts = 3, baseDelay = 300, signal } = {}) {
  for (let i = 0; i < attempts; i++) {
    try {
      return await fn({ signal });
    } catch (err) {
      const isLast = i === attempts - 1;
      if (isLast || !isRetryable(err) || signal?.aborted) throw err;

      const backoff = baseDelay * 2 ** i;
      await delay(backoff + Math.random() * backoff * 0.5, { signal });
    }
  }
}
```

Volunteer: only retry 5xx, 429 and network errors; jitter prevents synchronized retry storms; non-idempotent requests need an idempotency key. → [[24.3 - Retries Timeouts and Cancellation]]

**Concurrency limiter**

```js
async function mapLimit(items, limit, fn) {
  const results = new Array(items.length);
  const executing = new Set();

  for (const [index, item] of items.entries()) {
    const task = Promise.resolve()
      .then(() => fn(item, index))
      .then(value => { results[index] = value; });

    executing.add(task);
    task.finally(() => executing.delete(task));

    if (executing.size >= limit) await Promise.race(executing);
  }

  await Promise.all(executing);
  return results;
}
```

`Promise.race` over the in-flight set is the idiom for "wait until one slot frees". → [[04.7 - Async Concurrency Patterns]]

**Request deduplication**

```js
function dedupe(fn) {
  const inFlight = new Map();

  return (key, ...args) => {
    if (!inFlight.has(key)) {
      inFlight.set(key, Promise.resolve(fn(key, ...args)).finally(() => inFlight.delete(key)));
    }
    return inFlight.get(key);
  };
}
```

Works because a promise settles once and can have any number of consumers.

**Sequential task queue**

```js
function createQueue() {
  let tail = Promise.resolve();

  return (task) => {
    const result = tail.then(task);
    tail = result.catch(() => {});        // one failure must not break the chain
    return result;
  };
}
```

The `catch` on `tail` is essential — without it, one rejection skips every subsequent task.

## DOM and browser

**Event emitter**

```js
class EventEmitter {
  #listeners = new Map();

  on(event, fn) {
    if (!this.#listeners.has(event)) this.#listeners.set(event, new Set());
    this.#listeners.get(event).add(fn);
    return () => this.off(event, fn);            // return an unsubscriber
  }

  once(event, fn) {
    const wrapper = (...args) => { this.off(event, wrapper); fn(...args); };
    return this.on(event, wrapper);
  }

  off(event, fn) { this.#listeners.get(event)?.delete(fn); }

  emit(event, ...args) {
    for (const fn of [...(this.#listeners.get(event) ?? [])]) fn(...args);
  }
}
```

Copying the set before iterating prevents mutation-during-emit bugs. Returning an unsubscriber makes cleanup hard to forget.

**Event delegation**

```js
container.addEventListener("click", (e) => {
  const target = e.target.closest("[data-action]");
  if (!target || !container.contains(target)) return;

  handle(target.dataset.action, target.dataset.id);
});
```

The containment check is the detail most candidates omit — `closest` walks all the way up the document. → [[12.4 - Event Delegation]]

**Virtualized list**

```jsx
function VirtualList({ items, itemHeight = 40, height = 400 }) {
  const [scrollTop, setScrollTop] = useState(0);

  const start = Math.max(0, Math.floor(scrollTop / itemHeight) - 3);
  const visible = items.slice(start, start + Math.ceil(height / itemHeight) + 6);

  return (
    <div style={{ height, overflow: "auto" }} onScroll={e => setScrollTop(e.currentTarget.scrollTop)}>
      <div style={{ height: items.length * itemHeight, position: "relative" }}>
        {visible.map((item, i) => (
          <div key={item.id} style={{ position: "absolute", top: (start + i) * itemHeight, height: itemHeight }}>
            {item.name}
          </div>
        ))}
      </div>
    </div>
  );
}
```

Volunteer: overscan prevents blank rows, variable heights need measurement, and `aria-rowcount` is required for accessibility. → [[19.5 - Virtualization and Windowing]]

## React

**`useDebounce`**

```jsx
function useDebounce(value, delay) {
  const [debounced, setDebounced] = useState(value);

  useEffect(() => {
    const id = setTimeout(() => setDebounced(value), delay);
    return () => clearTimeout(id);
  }, [value, delay]);

  return debounced;
}
```

Debounce the **value**, not the handler — that sidesteps the function-identity problem entirely.

**`usePrevious`**

```jsx
function usePrevious(value) {
  const ref = useRef(undefined);
  useEffect(() => { ref.current = value; });
  return ref.current;
}
```

**`useLocalStorage`**

```jsx
function useLocalStorage(key, initial) {
  const [value, setValue] = useState(() => {
    try {
      const raw = localStorage.getItem(key);
      return raw ? JSON.parse(raw) : initial;
    } catch { return initial; }
  });

  useEffect(() => {
    try { localStorage.setItem(key, JSON.stringify(value)); } catch {}
  }, [key, value]);

  return [value, setValue];
}
```

Both try/catch blocks matter — storage throws in private browsing and on quota exceeded. Lazy initialization avoids reading storage on every render. → [[14.2 - LocalStorage and SessionStorage]]

**`useFetch` with cancellation**

```jsx
function useFetch(url) {
  const [state, setState] = useState({ status: "loading" });

  useEffect(() => {
    const controller = new AbortController();
    setState({ status: "loading" });

    (async () => {
      try {
        const res = await fetch(url, { signal: controller.signal });
        if (!res.ok) throw new Error(`HTTP ${res.status}`);
        setState({ status: "success", data: await res.json() });
      } catch (err) {
        if (err.name === "AbortError") return;
        setState({ status: "error", error: err });
      }
    })();

    return () => controller.abort();
  }, [url]);

  return state;
}
```

Volunteer what is still missing — caching, deduplication, retries, background refetch — as the reason to use a query library. → [[24.6 - TanStack Query Fundamentals]]

**Accessible modal**

```jsx
function Modal({ isOpen, onClose, children }) {
  const ref = useRef(null);

  useEffect(() => {
    const dialog = ref.current;
    isOpen ? dialog?.showModal() : dialog?.close();
  }, [isOpen]);

  return (
    <dialog
      ref={ref}
      onClose={onClose}
      onClick={e => { if (e.target === ref.current) onClose(); }}
    >
      {children}
    </dialog>
  );
}
```

`<dialog>` gives top-layer rendering, focus trapping, Escape handling and background inertness for free. Volunteer that as the reason not to hand-roll it. → [[22.2 - Portals]]

## How to perform in the round

**Clarify first.** Edge cases, expected inputs, whether cancellation or error handling is in scope. Two minutes here changes what you build.

**Write the working version, then improve.** A correct simple implementation plus a stated list of improvements beats a half-finished sophisticated one.

**Narrate the edge cases you are handling** — empty input, cancellation, cleanup, re-entrancy. That is what is actually being assessed.

**Volunteer the production concerns** even if not asked: cache eviction, cancellation, accessibility, cleanup. Naming what you would add signals experience more than the code itself.

**Test aloud.** Walk one input through, including an edge case.
