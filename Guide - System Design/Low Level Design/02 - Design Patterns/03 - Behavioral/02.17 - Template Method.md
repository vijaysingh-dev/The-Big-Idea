# Template Method Pattern

## At a Glance

|||
|---|---|
|**Type**|Behavioral|
|**Intent**|Define skeleton of an algorithm in a base class; subclasses fill in specific steps without changing the structure|
|**Frequency in Backend**|★★★★☆ — ETL pipelines, report generators, batch processors|

---

## Full Backend Example: Data Import Pipeline

```python
# data_importer.py
from abc import ABC, abstractmethod
from dataclasses import dataclass
from typing import List, Any
import time

@dataclass
class ImportResult:
    total_records: int
    imported: int
    failed: int
    skipped: int
    errors: list[str]

class DataImporter(ABC):
    """
    Template Method: defines fixed import workflow.
    Subclasses implement steps, not the workflow structure.
    """
    
    def import_data(self, source: str) -> ImportResult:
        """
        Fixed algorithm skeleton — subclasses cannot override this.
        Steps 2, 3, 4, 5 are customizable.
        """
        self._log(f"Starting import from {source}")
        start = time.monotonic()
        
        # Step 1: Fixed — open source
        raw_data = self._open_source(source)
        
        # Step 2: Hook — validate source
        self._validate_source(raw_data)
        
        # Step 3: Abstract — parse records
        records = self._parse(raw_data)
        
        # Step 4: Abstract — transform each record
        results = {"imported": 0, "failed": 0, "skipped": 0, "errors": []}
        for record in records:
            try:
                transformed = self._transform(record)
                if self._should_skip(transformed):
                    results["skipped"] += 1
                    continue
                self._persist(transformed)
                results["imported"] += 1
            except Exception as e:
                results["failed"] += 1
                results["errors"].append(f"Record {record}: {e}")
        
        # Step 5: Hook — post-import cleanup
        self._post_import(results)
        
        elapsed = time.monotonic() - start
        self._log(f"Import complete in {elapsed:.1f}s: {results}")
        
        return ImportResult(total_records=len(records), **results)
    
    # ── Abstract steps (must implement) ─────────────────────
    
    @abstractmethod
    def _parse(self, raw_data) -> List[Any]:
        """Parse raw source into list of raw records."""
        pass
    
    @abstractmethod
    def _transform(self, record) -> dict:
        """Transform one raw record into domain object dict."""
        pass
    
    @abstractmethod
    def _persist(self, record: dict):
        """Save transformed record to destination."""
        pass
    
    # ── Hook steps (optional override) ──────────────────────
    
    def _open_source(self, source: str):
        with open(source, 'r') as f:
            return f.read()
    
    def _validate_source(self, raw_data):
        """Override to add source-level validation."""
        pass
    
    def _should_skip(self, record: dict) -> bool:
        """Override to add skip conditions."""
        return False
    
    def _post_import(self, results: dict):
        """Override for cleanup, notifications, etc."""
        pass
    
    def _log(self, message: str):
        logger.info(f"[{self.__class__.__name__}] {message}")


# concrete_importers.py
import csv, json

class CsvUserImporter(DataImporter):
    """CSV → User import. Only implements the variable steps."""
    
    def __init__(self, user_repo):
        self._repo = user_repo
    
    def _parse(self, raw_data):
        reader = csv.DictReader(raw_data.splitlines())
        return list(reader)
    
    def _transform(self, record: dict) -> dict:
        return {
            "email": record["email"].strip().lower(),
            "name": f"{record['first_name']} {record['last_name']}".strip(),
            "role": record.get("role", "user"),
            "created_at": datetime.now()
        }
    
    def _should_skip(self, record: dict) -> bool:
        return not record.get("email")   # skip records with no email
    
    def _persist(self, record: dict):
        self._repo.upsert_by_email(record)
    
    def _post_import(self, results):
        if results["failed"] > 0:
            slack.alert(f"User import had {results['failed']} failures")


class JsonProductImporter(DataImporter):
    """JSON product catalog import."""
    
    def __init__(self, product_repo, category_service):
        self._repo = product_repo
        self._categories = category_service
    
    def _parse(self, raw_data):
        return json.loads(raw_data)["products"]
    
    def _validate_source(self, raw_data):
        data = json.loads(raw_data)
        if "products" not in data:
            raise ValueError("Invalid product catalog format — missing 'products' key")
    
    def _transform(self, record: dict) -> dict:
        return {
            "sku": record["sku"],
            "name": record["name"],
            "price": Decimal(str(record["price"])),
            "category_id": self._categories.get_or_create(record.get("category", "Uncategorized")).id,
            "active": record.get("active", True)
        }
    
    def _persist(self, record: dict):
        self._repo.upsert_by_sku(record)


# Usage — structure is identical; only behavior differs
csv_importer = CsvUserImporter(user_repo)
result = csv_importer.import_data("users_jan.csv")

json_importer = JsonProductImporter(product_repo, category_service)
result = json_importer.import_data("catalog_v2.json")
```

## When NOT to Use Template Method

- **Inheritance is the wrong tool** — composition (Strategy) is often more flexible
- **Subclasses need to reorder steps** — template method enforces order; Strategy is better
- **Skeleton is trivial** — just a plain function
