# State Pattern

## At a Glance

|||
|---|---|
|**Type**|Behavioral|
|**Intent**|Allow an object to alter its behavior when its internal state changes — the object appears to change its class|
|**Frequency in Backend**|★★★★☆ — order lifecycle, subscription states, document workflows|

---

## Strategy vs State

```
Strategy: context picks algorithm, algorithm doesn't change context's state
State:    state transitions the context to a new state — the state controls flow
```

---

## Full Backend Example: Subscription Lifecycle

```python
# subscription_state.py
from abc import ABC, abstractmethod
from dataclasses import dataclass
from datetime import datetime
from decimal import Decimal
from typing import Optional

class SubscriptionState(ABC):
    """
    Each concrete state implements what's allowed in that state.
    State objects hold the context reference so they can transition it.
    """
    
    @abstractmethod
    def activate(self, subscription: "Subscription"): pass
    
    @abstractmethod
    def suspend(self, subscription: "Subscription", reason: str): pass
    
    @abstractmethod
    def cancel(self, subscription: "Subscription", reason: str): pass
    
    @abstractmethod
    def renew(self, subscription: "Subscription"): pass
    
    @abstractmethod
    def status_name(self) -> str: pass
    
    def can_access_features(self) -> bool:
        return False


class TrialState(SubscriptionState):
    """Free trial — limited time, can convert to active."""
    
    def activate(self, sub):
        """Convert trial to paid subscription."""
        sub._state = ActiveState()
        sub._current_period_start = datetime.now()
        sub._events.append(SubscriptionActivated(sub.id))
    
    def suspend(self, sub, reason):
        raise InvalidStateTransition("Cannot suspend a trial subscription")
    
    def cancel(self, sub, reason):
        sub._state = CancelledState()
        sub._cancelled_at = datetime.now()
        sub._cancellation_reason = reason
        sub._events.append(SubscriptionCancelled(sub.id, reason))
    
    def renew(self, sub):
        raise InvalidStateTransition("Cannot renew a trial — activate first")
    
    def status_name(self): return "trial"
    def can_access_features(self): return True   # trial has access


class ActiveState(SubscriptionState):
    """Paid and current — full access."""
    
    def activate(self, sub):
        raise InvalidStateTransition("Subscription is already active")
    
    def suspend(self, sub, reason):
        sub._state = SuspendedState()
        sub._suspended_at = datetime.now()
        sub._suspension_reason = reason
        sub._events.append(SubscriptionSuspended(sub.id, reason))
    
    def cancel(self, sub, reason):
        sub._state = CancelledState()
        sub._cancelled_at = datetime.now()
        sub._cancellation_reason = reason
        sub._events.append(SubscriptionCancelled(sub.id, reason))
    
    def renew(self, sub):
        """Extend the current period."""
        from dateutil.relativedelta import relativedelta
        sub._current_period_end = sub._current_period_end + relativedelta(months=1)
        sub._events.append(SubscriptionRenewed(sub.id))
    
    def status_name(self): return "active"
    def can_access_features(self): return True


class SuspendedState(SubscriptionState):
    """Payment failed or manually suspended — no access."""
    
    def activate(self, sub):
        """Re-activate after resolving suspension."""
        sub._state = ActiveState()
        sub._suspended_at = None
        sub._suspension_reason = None
        sub._events.append(SubscriptionReactivated(sub.id))
    
    def suspend(self, sub, reason):
        raise InvalidStateTransition("Already suspended")
    
    def cancel(self, sub, reason):
        sub._state = CancelledState()
        sub._cancelled_at = datetime.now()
        sub._cancellation_reason = reason
        sub._events.append(SubscriptionCancelled(sub.id, reason))
    
    def renew(self, sub):
        raise InvalidStateTransition("Cannot renew suspended subscription — resolve first")
    
    def status_name(self): return "suspended"
    def can_access_features(self): return False


class CancelledState(SubscriptionState):
    """Terminal state — no transitions allowed."""
    
    def activate(self, sub):
        raise InvalidStateTransition("Cannot activate a cancelled subscription")
    
    def suspend(self, sub, reason):
        raise InvalidStateTransition("Cannot suspend a cancelled subscription")
    
    def cancel(self, sub, reason):
        raise InvalidStateTransition("Already cancelled")
    
    def renew(self, sub):
        raise InvalidStateTransition("Cannot renew cancelled subscription")
    
    def status_name(self): return "cancelled"
    def can_access_features(self): return False


# subscription.py — Context
class Subscription:
    """
    Context: delegates all behavior to current state.
    Holds state as an object, not a string enum.
    """
    
    def __init__(self, subscription_id, plan, user_id):
        self.id = subscription_id
        self.plan = plan
        self.user_id = user_id
        self._state: SubscriptionState = TrialState()
        self._current_period_start = datetime.now()
        self._current_period_end = None
        self._suspended_at = None
        self._cancelled_at = None
        self._suspension_reason = None
        self._cancellation_reason = None
        self._events = []
    
    # Delegate all operations to current state
    def activate(self):             self._state.activate(self)
    def suspend(self, reason: str): self._state.suspend(self, reason)
    def cancel(self, reason: str):  self._state.cancel(self, reason)
    def renew(self):                self._state.renew(self)
    
    @property
    def status(self) -> str:
        return self._state.status_name()
    
    @property
    def can_access(self) -> bool:
        return self._state.can_access_features()
    
    def get_events(self):
        return list(self._events)
    
    def clear_events(self):
        self._events.clear()


# State diagram
"""
   ┌──────────────────────────────────────────────────┐
   │                                                  │
   ▼                                                  │
 TRIAL ──activate()──▶ ACTIVE  ──renew()──────────────┘
   │                    │   │
   │                    │   └──suspend()──▶ SUSPENDED ──activate()──▶ ACTIVE
   │                    │                       │
   └──cancel()──────────┴──cancel()─────────────┤
                                                ▼
                                          CANCELLED (terminal)
"""

# Usage
sub = Subscription(id=uuid4(), plan="pro", user_id=user_id)
print(sub.status)       # "trial"
print(sub.can_access)   # True

sub.activate()          # trial → active
print(sub.status)       # "active"

sub.suspend("payment_failed")  # active → suspended
print(sub.can_access)          # False

sub.activate()          # suspended → active (re-activated)
sub.cancel("user_request")     # active → cancelled

sub.renew()             # raises InvalidStateTransition
```

## State vs Enum + If/Elif

```python
# ❌ Enum approach — adding a new state means editing this function
def process_action(sub, action):
    if sub.status == "trial":
        if action == "activate": ...
        elif action == "cancel": ...
    elif sub.status == "active":
        if action == "suspend": ...
        elif action == "cancel": ...
        elif action == "renew": ...
    # Adding "grace_period" state = edit this entire function

# ✅ State pattern — adding a state = add a new class
class GracePeriodState(SubscriptionState):
    def activate(self, sub): ...  # all behavior in one place
    def suspend(self, sub, reason): ...
    ...
```

## Real-World Backend Uses

|Entity|States|
|---|---|
|Order|Draft → Confirmed → Paid → Shipped → Delivered / Cancelled|
|Subscription|Trial → Active → Suspended → Cancelled|
|Job/Task|Pending → Running → Completed / Failed / Retrying|
|Payment|Pending → Authorized → Captured / Refunded / Failed|
|Document|Draft → Review → Approved → Published / Rejected|
|Ticket|Open → In Progress → Resolved → Closed / Reopened|

---

## When NOT to Use State

- **2 states only** — a boolean flag is clearer
- **States share most behavior** — the pattern adds boilerplate without benefit
- **No transitions** — just use an enum

---

## Tradeoffs

|Pro|Con|
|---|---|
|Eliminates large if/elif on status|Many state classes|
|Transitions enforced and tested per-state|State must hold context reference (coupling)|
|Easy to add new state (OCP)|Distributed logic — need to read all states to understand flow|
|Business rules per-state are explicit||
