# API Design

## REST Principles

```
Resource-based URLs:      /orders, /orders/{id}, /users/{id}/orders
HTTP verbs for actions:   GET (read), POST (create), PUT/PATCH (update), DELETE
Stateless:                Server doesn't hold client state between requests
Uniform interface:        Consistent patterns across all endpoints
```

---

## URL Design

```
# Resources: plural nouns
GET    /orders                    # list orders
POST   /orders                    # create order
GET    /orders/{id}               # get specific order
PUT    /orders/{id}               # replace order (full update)
PATCH  /orders/{id}               # partial update
DELETE /orders/{id}               # delete order

# Sub-resources for relationships
GET    /orders/{id}/items         # items of a specific order
POST   /orders/{id}/items         # add item to order
DELETE /orders/{id}/items/{item}  # remove item

# Actions as sub-resources (not verbs in URL)
POST /orders/{id}/cancel          # action: cancel
POST /orders/{id}/ship            # action: ship
POST /payments/{id}/refund        # action: refund

# Filtering, sorting, pagination as query params
GET /orders?status=pending&user_id=123&sort=created_at&order=desc&page=2&limit=20
```

---

## HTTP Status Codes — Get These Right

```
2xx Success:
  200 OK              — successful read, update, action
  201 Created         — resource created (include Location header)
  202 Accepted        — async processing started (not done yet)
  204 No Content      — success with no response body (DELETE)

4xx Client errors:
  400 Bad Request     — malformed syntax, missing required field
  401 Unauthorized    — not authenticated (missing/invalid token)
  403 Forbidden       — authenticated but not authorized
  404 Not Found       — resource doesn't exist
  405 Method Not Allowed — POST on read-only endpoint
  409 Conflict        — duplicate resource, state conflict
  410 Gone            — permanently deleted
  422 Unprocessable   — syntactically valid but semantically wrong (validation failure)
  429 Too Many Requests — rate limited

5xx Server errors:
  500 Internal Server Error — unexpected server failure
  502 Bad Gateway           — upstream service failure
  503 Service Unavailable   — overloaded, maintenance
  504 Gateway Timeout       — upstream timeout
```

## Pagination Strategies

```python
# Offset pagination (simple but has issues at scale)
GET /orders?page=2&limit=20
# SQL: LIMIT 20 OFFSET 20
# Problems:
#   - If items added during iteration, items may shift → skip or duplicate
#   - Performance degrades at high offsets (DB must scan to offset)

# Cursor pagination (recommended for large datasets)
GET /orders?cursor=eyJpZCI6MTIzfQ&limit=20
# Cursor = opaque base64 of last seen item's sort key (e.g., id or created_at)
# SQL: WHERE id > :cursor_id ORDER BY id LIMIT 20
# Problems: can't jump to page N directly

class CursorPaginator:
    def paginate(self, query, cursor: Optional[str], limit: int) -> dict:
        if cursor:
            decoded = json.loads(base64.b64decode(cursor).decode())
            query = query.filter(Order.id > decoded["id"])
        
        items = query.order_by(Order.id).limit(limit + 1).all()
        
        has_more = len(items) > limit
        if has_more:
            items = items[:limit]
        
        next_cursor = None
        if has_more:
            next_cursor = base64.b64encode(
                json.dumps({"id": str(items[-1].id)}).encode()
            ).decode()
        
        return {
            "data": items,
            "next_cursor": next_cursor,
            "has_more": has_more
        }
```

## API Versioning

```python
# 1. URL versioning (most common, most visible)
GET /v1/orders
GET /v2/orders

# 2. Header versioning (cleaner URLs, less visible)
GET /orders
Accept: application/vnd.company.v2+json

# 3. Query param (least preferred)
GET /orders?version=2

# Deprecation strategy
# When introducing v2, v1 stays alive for 6+ months
# Response headers signal deprecation:
Deprecation: true
Sunset: Sat, 31 Dec 2024 23:59:59 GMT
Link: <https://api.company.com/v2/orders>; rel="successor-version"
```

## Rate Limiting Headers

```python
# Standard headers for rate limit communication
X-RateLimit-Limit:     100        # requests allowed per window
X-RateLimit-Remaining: 87         # requests left in current window
X-RateLimit-Reset:     1700000000 # unix timestamp when window resets

# When limited:
HTTP/1.1 429 Too Many Requests
Retry-After: 30                   # seconds until retry safe

class RateLimitMiddleware:
    def process(self, request, next_handler):
        key = self._get_key(request)       # by user, IP, or API key
        result = self._limiter.check(key, limit=100, window=60)
        
        response = next_handler(request)
        
        response.headers["X-RateLimit-Limit"] = str(result.limit)
        response.headers["X-RateLimit-Remaining"] = str(result.remaining)
        response.headers["X-RateLimit-Reset"] = str(result.reset_at)
        
        if result.exceeded:
            response.status_code = 429
            response.headers["Retry-After"] = str(result.retry_after_seconds)
        
        return response
```

## Idempotency Keys (for POST)

```python
# Client sends Idempotency-Key header for non-idempotent operations
POST /payments
Idempotency-Key: client-generated-uuid-123
Content-Type: application/json
{"amount": 9999, "currency": "INR"}

# Server behavior:
# - First request: process and store response with key
# - Duplicate request: return stored response (don't reprocess)
# - Different body, same key: return 422 (key collision)

class IdempotencyMiddleware:
    def process(self, request, next_handler):
        if request.method != "POST":
            return next_handler(request)
        
        key = request.headers.get("Idempotency-Key")
        if not key:
            return next_handler(request)
        
        # Check for stored response
        stored = self._store.get(key)
        if stored:
            return HttpResponse(
                status_code=stored["status"],
                body=stored["body"],
                headers={**stored["headers"], "X-Idempotent-Replayed": "true"}
            )
        
        response = next_handler(request)
        
        # Store response for 24 hours
        self._store.set(key, {
            "status": response.status_code,
            "body": response.body,
            "headers": dict(response.headers)
        }, ttl=86400)
        
        return response
```

## Webhooks

```python
# Outbound webhook delivery
class WebhookDeliveryService:
    def deliver(self, event: dict, webhook_url: str, secret: str):
        payload = json.dumps(event).encode()
        
        # HMAC signature so receiver can verify it's us
        sig = hmac.new(secret.encode(), payload, hashlib.sha256).hexdigest()
        
        response = requests.post(
            webhook_url,
            data=payload,
            headers={
                "Content-Type": "application/json",
                "X-Webhook-Signature": f"sha256={sig}",
                "X-Webhook-Event": event["type"],
                "X-Webhook-Delivery": str(uuid4()),
            },
            timeout=10
        )
        
        if not response.ok:
            raise WebhookDeliveryError(f"Webhook failed: {response.status_code}")

# Inbound webhook verification
class WebhookReceiver:
    def verify_and_parse(self, payload: bytes, signature_header: str, secret: str) -> dict:
        expected_sig = hmac.new(secret.encode(), payload, hashlib.sha256).hexdigest()
        received_sig = signature_header.replace("sha256=", "")
        
        if not hmac.compare_digest(expected_sig, received_sig):
            raise InvalidWebhookSignature("Webhook signature verification failed")
        
        return json.loads(payload)
```
