# Mediator Pattern

## At a Glance

|||
|---|---|
|**Type**|Behavioral|
|**Intent**|Define an object that encapsulates how a set of objects interact — promotes loose coupling by preventing objects from referring to each other explicitly|
|**Also Known As**|Command Bus, Message Dispatcher|
|**Frequency in Backend**|★★★☆☆ — CQRS command/query bus, event dispatchers, chat systems|

---

## The Problem It Solves

Many components need to communicate. Direct connections create a mesh of dependencies — each component knows about all others. Adding a new component requires updating all existing ones.

```
Without Mediator (mesh):          With Mediator (hub-and-spoke):
  A ←──────→ B                      A ──→ Mediator ←── B
  ↕    ╲  /  ↕                      C ──→           ←── D
  C ←──────→ D                    Components only know Mediator
  [N×(N-1) connections]           [N connections total]
```

## Full Backend Example: Command/Query Bus (MediatR-style)

```python
# mediator.py
from abc import ABC, abstractmethod
from typing import TypeVar, Generic, Type, Any, Callable
from dataclasses import dataclass
import time, logging

TRequest = TypeVar("TRequest")
TResponse = TypeVar("TResponse")

logger = logging.getLogger(__name__)


class Request(ABC, Generic[TResponse]):
    """Marker base for all commands and queries."""
    pass


class RequestHandler(ABC, Generic[TRequest, TResponse]):
    @abstractmethod
    def handle(self, request: TRequest) -> TResponse: pass


class PipelineBehavior(ABC):
    """Middleware for the mediator pipeline (logging, validation, caching)."""
    
    @abstractmethod
    def handle(self, request: Request, next_handler: Callable) -> Any: pass


class Mediator:
    """
    Central dispatcher.
    Controllers send requests here; handlers are registered separately.
    Eliminates direct service-to-service dependencies in the application layer.
    """
    
    def __init__(self):
        self._handlers: dict[Type, RequestHandler] = {}
        self._behaviors: list[PipelineBehavior] = []
    
    def register_handler(self, request_type: Type, handler: RequestHandler):
        if request_type in self._handlers:
            raise ValueError(f"Handler already registered for {request_type.__name__}")
        self._handlers[request_type] = handler
    
    def add_behavior(self, behavior: PipelineBehavior):
        """Behaviors run around every request (like middleware)."""
        self._behaviors.append(behavior)
    
    def send(self, request: Request) -> Any:
        handler = self._handlers.get(type(request))
        if not handler:
            raise ValueError(f"No handler registered for {type(request).__name__}")
        
        # Build pipeline: behaviors wrap the core handler
        def core():
            return handler.handle(request)
        
        pipeline = core
        for behavior in reversed(self._behaviors):
            outer = pipeline
            pipeline = lambda b=behavior, nxt=outer: b.handle(request, nxt)
        
        return pipeline()


# pipeline_behaviors.py
class LoggingBehavior(PipelineBehavior):
    """Log every request and its duration."""
    
    def handle(self, request: Request, next_handler: Callable) -> Any:
        start = time.monotonic()
        request_name = type(request).__name__
        logger.info(f"→ Handling {request_name}")
        
        try:
            result = next_handler()
            elapsed = (time.monotonic() - start) * 1000
            logger.info(f"← {request_name} completed in {elapsed:.1f}ms")
            return result
        except Exception as e:
            elapsed = (time.monotonic() - start) * 1000
            logger.error(f"← {request_name} failed after {elapsed:.1f}ms: {e}")
            raise


class ValidationBehavior(PipelineBehavior):
    """Validate request before handling."""
    
    def __init__(self, validators: dict[Type, list]):
        self._validators = validators
    
    def handle(self, request: Request, next_handler: Callable) -> Any:
        validators = self._validators.get(type(request), [])
        errors = []
        for validator in validators:
            errors.extend(validator.validate(request))
        
        if errors:
            raise ValidationError(errors)
        
        return next_handler()


class CachingBehavior(PipelineBehavior):
    """Cache query results (skip commands)."""
    
    def __init__(self, cache, cacheable_types: set[Type], ttl: int = 300):
        self._cache = cache
        self._cacheable = cacheable_types
        self._ttl = ttl
    
    def handle(self, request: Request, next_handler: Callable) -> Any:
        if type(request) not in self._cacheable:
            return next_handler()
        
        import hashlib, json
        cache_key = f"mediator:{type(request).__name__}:{hashlib.md5(json.dumps(vars(request), sort_keys=True, default=str).encode()).hexdigest()}"
        
        cached = self._cache.get(cache_key)
        if cached is not None:
            return cached
        
        result = next_handler()
        self._cache.set(cache_key, result, self._ttl)
        return result


# commands_and_queries.py
from uuid import UUID
from dataclasses import dataclass

# Commands (write operations)
@dataclass
class PlaceOrderCommand(Request["Order"]):
    user_id: UUID
    items: list
    payment_method_id: str
    shipping_address_id: str
    idempotency_key: str

@dataclass
class CancelOrderCommand(Request[bool]):
    order_id: UUID
    user_id: UUID
    reason: str

@dataclass
class UpdateUserProfileCommand(Request["User"]):
    user_id: UUID
    name: str
    phone: str

# Queries (read operations)
@dataclass
class GetOrderQuery(Request["Order"]):
    order_id: UUID
    user_id: UUID   # for authorization

@dataclass
class ListUserOrdersQuery(Request[list]):
    user_id: UUID
    status_filter: str = None
    page: int = 1
    page_size: int = 20

@dataclass
class GetProductQuery(Request["Product"]):
    product_id: UUID


# handlers.py
class PlaceOrderHandler(RequestHandler[PlaceOrderCommand, "Order"]):
    def __init__(self, order_service, inventory_service, payment_service):
        self._orders = order_service
        self._inventory = inventory_service
        self._payments = payment_service
    
    def handle(self, cmd: PlaceOrderCommand) -> "Order":
        return self._orders.place(cmd)


class GetOrderHandler(RequestHandler[GetOrderQuery, "Order"]):
    def __init__(self, order_repo, access_policy):
        self._repo = order_repo
        self._policy = access_policy
    
    def handle(self, query: GetOrderQuery) -> "Order":
        order = self._repo.get_or_raise(query.order_id)
        if not self._policy.can_view(query.user_id, order):
            raise PermissionDenied(f"Cannot access order {query.order_id}")
        return order


class ListUserOrdersHandler(RequestHandler[ListUserOrdersQuery, list]):
    def __init__(self, order_read_repo):
        self._repo = order_read_repo
    
    def handle(self, query: ListUserOrdersQuery) -> list:
        return self._repo.find_by_user(
            user_id=query.user_id,
            status=query.status_filter,
            page=query.page,
            page_size=query.page_size
        )


# wiring.py — composition root
def build_mediator(container) -> Mediator:
    mediator = Mediator()
    
    # Register behaviors (run for every request)
    mediator.add_behavior(LoggingBehavior())
    mediator.add_behavior(ValidationBehavior(validators={
        PlaceOrderCommand: [OrderValidator()],
    }))
    mediator.add_behavior(CachingBehavior(
        cache=container.cache,
        cacheable_types={GetOrderQuery, ListUserOrdersQuery, GetProductQuery},
        ttl=60
    ))
    
    # Register handlers
    mediator.register_handler(PlaceOrderCommand, PlaceOrderHandler(
        container.order_service,
        container.inventory_service,
        container.payment_service
    ))
    mediator.register_handler(CancelOrderCommand, CancelOrderHandler(container.order_service))
    mediator.register_handler(GetOrderQuery, GetOrderHandler(container.order_repo, container.access_policy))
    mediator.register_handler(ListUserOrdersQuery, ListUserOrdersHandler(container.order_read_repo))
    
    return mediator


# controller.py — clean, no direct service dependencies
class OrderController:
    def __init__(self, mediator: Mediator):
        self._mediator = mediator
    
    def place_order(self, request_body: dict, current_user):
        order = self._mediator.send(PlaceOrderCommand(
            user_id=current_user.id,
            items=request_body["items"],
            payment_method_id=request_body["payment_method_id"],
            shipping_address_id=request_body["shipping_address_id"],
            idempotency_key=request_body["idempotency_key"]
        ))
        return {"order_id": str(order.id)}, 201
    
    def get_order(self, order_id: str, current_user):
        order = self._mediator.send(GetOrderQuery(
            order_id=UUID(order_id),
            user_id=current_user.id
        ))
        return order.to_dict()
    
    def list_orders(self, current_user, status: str = None, page: int = 1):
        orders = self._mediator.send(ListUserOrdersQuery(
            user_id=current_user.id,
            status_filter=status,
            page=page
        ))
        return [o.to_dict() for o in orders]
```

## Mediator Benefits in a Layered App

```
Without Mediator:
  OrderController → OrderService → InventoryService
                  → PaymentService
                  → NotificationService
  [Controller has 4 direct dependencies]

With Mediator:
  OrderController → Mediator (single dependency)
  Mediator → PlaceOrderHandler → OrderService
           → GetOrderHandler   → OrderRepo
  [Controller knows only Mediator]
```

---

## When NOT to Use Mediator

- **Simple CRUD** — adds unnecessary indirection
- **Only 2-3 components** — direct wiring is clearer
- **Performance-critical hot path** — the dispatch overhead, though small, exists

---

## Tradeoffs

|Pro|Con|
|---|---|
|Controllers have single dependency|Mediator itself can become complex|
|Cross-cutting concerns centralized (logging, caching)|Indirect dispatch — harder to trace|
|Easy to add new handlers (OCP)|If handler not registered, silent failure risk|
|Testable handlers in isolation|Wiring at composition root needs discipline|
