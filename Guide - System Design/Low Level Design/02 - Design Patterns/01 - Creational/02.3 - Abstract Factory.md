# Abstract Factory Pattern

## At a Glance

|||
|---|---|
|**Type**|Creational|
|**Intent**|Provide an interface for creating **families of related objects** without specifying their concrete classes|
|**Frequency in Backend**|★★★☆☆ — multi-provider infra, testing setups|

---

## Factory Method vs Abstract Factory

```
Factory Method:   Creates ONE type of object, multiple variants
                  e.g., "give me a payment gateway"

Abstract Factory: Creates a FAMILY of related objects, consistently
                  e.g., "give me a full infrastructure set:
                         DB + Cache + Queue that all work together"
```

---

## Full Backend Example: Infrastructure Provider Factory

```python
# infrastructure_interfaces.py
from abc import ABC, abstractmethod
from typing import Any, Optional

class DatabaseClient(ABC):
    @abstractmethod
    def execute(self, query: str, params: tuple = ()) -> list[dict]: pass
    
    @abstractmethod
    def execute_one(self, query: str, params: tuple = ()) -> Optional[dict]: pass
    
    @abstractmethod
    def begin_transaction(self): pass

class CacheClient(ABC):
    @abstractmethod
    def get(self, key: str) -> Optional[Any]: pass
    
    @abstractmethod
    def set(self, key: str, value: Any, ttl_seconds: int = 300): pass
    
    @abstractmethod
    def delete(self, key: str): pass
    
    @abstractmethod
    def exists(self, key: str) -> bool: pass

class QueueClient(ABC):
    @abstractmethod
    def publish(self, topic: str, message: dict): pass
    
    @abstractmethod
    def subscribe(self, topic: str, handler): pass


# abstract_factory.py
class InfrastructureFactory(ABC):
    """Abstract Factory — creates a consistent family of infra clients."""
    
    @abstractmethod
    def create_db(self) -> DatabaseClient: pass
    
    @abstractmethod
    def create_cache(self) -> CacheClient: pass
    
    @abstractmethod
    def create_queue(self) -> QueueClient: pass


# production_factory.py — PostgreSQL + Redis + Kafka
class PostgresDatabaseClient(DatabaseClient):
    def __init__(self, connection_string: str):
        import psycopg2
        self._conn = psycopg2.connect(connection_string)
    
    def execute(self, query, params=()):
        with self._conn.cursor() as cur:
            cur.execute(query, params)
            return [dict(row) for row in cur.fetchall()]
    
    def execute_one(self, query, params=()):
        results = self.execute(query, params)
        return results[0] if results else None
    
    def begin_transaction(self):
        return self._conn  # context manager

class RedisCacheClient(CacheClient):
    def __init__(self, redis_url: str):
        import redis
        self._client = redis.from_url(redis_url)
    
    def get(self, key):
        import pickle
        val = self._client.get(key)
        return pickle.loads(val) if val else None
    
    def set(self, key, value, ttl_seconds=300):
        import pickle
        self._client.setex(key, ttl_seconds, pickle.dumps(value))
    
    def delete(self, key): self._client.delete(key)
    def exists(self, key): return self._client.exists(key) > 0

class KafkaQueueClient(QueueClient):
    def __init__(self, bootstrap_servers: str):
        from kafka import KafkaProducer
        self._producer = KafkaProducer(bootstrap_servers=bootstrap_servers)
    
    def publish(self, topic, message):
        import json
        self._producer.send(topic, json.dumps(message).encode())
    
    def subscribe(self, topic, handler):
        from kafka import KafkaConsumer
        consumer = KafkaConsumer(topic)
        for msg in consumer:
            handler(msg.value)


class ProductionInfraFactory(InfrastructureFactory):
    def __init__(self, config: AppConfig):
        self._config = config
    
    def create_db(self) -> DatabaseClient:
        return PostgresDatabaseClient(self._config.database_url)
    
    def create_cache(self) -> CacheClient:
        return RedisCacheClient(self._config.redis_url)
    
    def create_queue(self) -> QueueClient:
        return KafkaQueueClient(self._config.kafka_brokers)


# test_factory.py — Fast in-memory implementations for testing
class InMemoryDatabaseClient(DatabaseClient):
    def __init__(self):
        self._tables: dict[str, list[dict]] = {}
    
    def execute(self, query, params=()):
        # Minimal SQL simulation for tests
        return []
    
    def execute_one(self, query, params=()):
        return None
    
    def begin_transaction(self):
        return self  # no-op context manager

class InMemoryCacheClient(CacheClient):
    def __init__(self):
        self._store: dict = {}
    
    def get(self, key): return self._store.get(key)
    def set(self, key, value, ttl_seconds=300): self._store[key] = value
    def delete(self, key): self._store.pop(key, None)
    def exists(self, key): return key in self._store

class InMemoryQueueClient(QueueClient):
    def __init__(self):
        self._handlers: dict = {}
    
    def publish(self, topic, message):
        handler = self._handlers.get(topic)
        if handler:
            handler(message)
    
    def subscribe(self, topic, handler):
        self._handlers[topic] = handler


class TestInfraFactory(InfrastructureFactory):
    """Full in-memory factory — no external deps, instant tests."""
    
    def create_db(self) -> DatabaseClient:
        return InMemoryDatabaseClient()
    
    def create_cache(self) -> CacheClient:
        return InMemoryCacheClient()
    
    def create_queue(self) -> QueueClient:
        return InMemoryQueueClient()


# app.py — service layer uses factory, never concrete types
class OrderProcessingService:
    def __init__(self, infra: InfrastructureFactory):
        self._db = infra.create_db()
        self._cache = infra.create_cache()
        self._queue = infra.create_queue()
    
    def process_order(self, order_id: str):
        # Works identically in prod (Postgres+Redis+Kafka)
        # and in tests (all in-memory)
        cached = self._cache.get(f"order:{order_id}")
        if cached:
            return cached
        
        order = self._db.execute_one("SELECT * FROM orders WHERE id = %s", (order_id,))
        self._cache.set(f"order:{order_id}", order, ttl_seconds=60)
        self._queue.publish("order.viewed", {"order_id": order_id})
        return order


# Wiring
if os.environ.get("ENV") == "test":
    factory = TestInfraFactory()
else:
    factory = ProductionInfraFactory(config)

service = OrderProcessingService(factory)
```

## Real-World Backend Uses

|Factory Creates|Family Members|
|---|---|
|Cloud infrastructure|DB + Storage + Queue (AWS vs GCP vs Azure)|
|Test vs Prod|In-memory vs Real implementations|
|SQL vs NoSQL|Different DB + ORM + migration tool sets|
|Notification stack|Email sender + template engine + tracker|

---

## When NOT to Use

- **Only one product in the family** — use Factory Method instead
- **Family members don't need to be consistent** — no relationship between members
- **Simple creation** — over-engineering if you don't swap families

---

## Tradeoffs

|Pro|Con|
|---|---|
|Consistent families — no mixing incompatible products|Hard to extend: adding a new product type requires updating ALL factories|
|Easy to swap entire environment (prod/test/staging)|More classes than Factory Method|
|Products guaranteed compatible|Family constraints can be too rigid|
