# Case Study: Ride-Sharing Dispatch

**Difficulty:** Hard | **Patterns:** State, Strategy, Observer, Command, Factory

---

## Step 1: Requirements

### Functional

- Rider requests a ride (pickup + dropoff + type)
- System finds and assigns nearest available driver
- Driver accepts or rejects; auto-reassign on rejection/timeout
- Real-time tracking (not in LLD scope — use WebSocket/pub-sub)
- Dynamic pricing (surge pricing based on supply/demand)
- Multiple ride types: Economy, Premium, XL, Auto
- Cancel flow: rider or driver cancels
- Completed ride triggers payment

### Non-Functional

- Driver match within 10 seconds of request
- Consistent state under concurrent requests
- Location data freshness: <30 seconds
- 99.9% availability

---

## Step 2: State Machines

```
Ride lifecycle:
  REQUESTED ──[driver_found]──▶ DRIVER_ASSIGNED
             ──[no_driver/timeout]──▶ CANCELLED
  
  DRIVER_ASSIGNED ──[driver_accepts]──▶ DRIVER_EN_ROUTE
                  ──[driver_rejects/timeout]──▶ REQUESTED (retry matching)
  
  DRIVER_EN_ROUTE ──[driver_arrives]──▶ DRIVER_ARRIVED
                  ──[cancelled]──▶ CANCELLED
  
  DRIVER_ARRIVED ──[trip_started]──▶ IN_PROGRESS
  
  IN_PROGRESS ──[trip_completed]──▶ COMPLETED
              ──[emergency]──▶ CANCELLED
  
  COMPLETED ── (terminal)
  CANCELLED ── (terminal)

Driver availability:
  OFFLINE ←──── AVAILABLE ──[assigned]──▶ ON_TRIP
                            ◀──[trip_complete]──
```

## Step 3: Full Design

```python
# models.py
from enum import Enum
from dataclasses import dataclass, field
from uuid import UUID, uuid4
from datetime import datetime
from decimal import Decimal
from typing import Optional, List

class RideStatus(Enum):
    REQUESTED = "REQUESTED"
    DRIVER_ASSIGNED = "DRIVER_ASSIGNED"
    DRIVER_EN_ROUTE = "DRIVER_EN_ROUTE"
    DRIVER_ARRIVED = "DRIVER_ARRIVED"
    IN_PROGRESS = "IN_PROGRESS"
    COMPLETED = "COMPLETED"
    CANCELLED = "CANCELLED"

class DriverStatus(Enum):
    OFFLINE = "OFFLINE"
    AVAILABLE = "AVAILABLE"
    ON_TRIP = "ON_TRIP"

class RideType(Enum):
    ECONOMY = "ECONOMY"
    PREMIUM = "PREMIUM"
    XL = "XL"
    AUTO = "AUTO"

@dataclass(frozen=True)
class Location:
    latitude: float
    longitude: float
    
    def distance_km(self, other: "Location") -> float:
        """Haversine distance."""
        import math
        R = 6371
        lat1, lon1 = math.radians(self.latitude), math.radians(self.longitude)
        lat2, lon2 = math.radians(other.latitude), math.radians(other.longitude)
        dlat, dlon = lat2 - lat1, lon2 - lon1
        a = math.sin(dlat/2)**2 + math.cos(lat1)*math.cos(lat2)*math.sin(dlon/2)**2
        return R * 2 * math.asin(math.sqrt(a))

@dataclass
class Fare:
    base_fare: Decimal
    distance_fare: Decimal
    time_fare: Decimal
    surge_multiplier: Decimal
    
    @property
    def total(self) -> Decimal:
        subtotal = self.base_fare + self.distance_fare + self.time_fare
        return (subtotal * self.surge_multiplier).quantize(Decimal("0.01"))


# ride.py — Aggregate with state machine
VALID_RIDE_TRANSITIONS = {
    RideStatus.REQUESTED:        {RideStatus.DRIVER_ASSIGNED, RideStatus.CANCELLED},
    RideStatus.DRIVER_ASSIGNED:  {RideStatus.DRIVER_EN_ROUTE, RideStatus.REQUESTED, RideStatus.CANCELLED},
    RideStatus.DRIVER_EN_ROUTE:  {RideStatus.DRIVER_ARRIVED, RideStatus.CANCELLED},
    RideStatus.DRIVER_ARRIVED:   {RideStatus.IN_PROGRESS, RideStatus.CANCELLED},
    RideStatus.IN_PROGRESS:      {RideStatus.COMPLETED, RideStatus.CANCELLED},
    RideStatus.COMPLETED:        set(),
    RideStatus.CANCELLED:        set(),
}

class Ride:
    def __init__(self, ride_id: UUID, rider_id: UUID, pickup: Location,
                 dropoff: Location, ride_type: RideType):
        self.id = ride_id
        self.rider_id = rider_id
        self.pickup = pickup
        self.dropoff = dropoff
        self.ride_type = ride_type
        self._status = RideStatus.REQUESTED
        self._driver_id: Optional[UUID] = None
        self._assignment_expires_at: Optional[datetime] = None
        self._fare: Optional[Fare] = None
        self._started_at: Optional[datetime] = None
        self._completed_at: Optional[datetime] = None
        self._cancellation_reason: Optional[str] = None
        self._match_attempts: int = 0
        self._events = []
        self.created_at = datetime.now()
    
    @property
    def status(self): return self._status
    @property
    def driver_id(self): return self._driver_id
    
    def assign_driver(self, driver_id: UUID, accept_timeout_seconds: int = 30):
        self._transition(RideStatus.DRIVER_ASSIGNED)
        self._driver_id = driver_id
        self._match_attempts += 1
        self._assignment_expires_at = datetime.now().replace(
            second=datetime.now().second + accept_timeout_seconds
        )
        self._events.append(DriverAssigned(ride_id=self.id, driver_id=driver_id))
    
    def driver_accepted(self):
        self._transition(RideStatus.DRIVER_EN_ROUTE)
        self._assignment_expires_at = None
        self._events.append(RideDriverAccepted(ride_id=self.id, driver_id=self._driver_id))
    
    def driver_rejected(self):
        """Driver rejected — go back to REQUESTED for re-matching."""
        self._transition(RideStatus.REQUESTED)
        old_driver = self._driver_id
        self._driver_id = None
        self._assignment_expires_at = None
        self._events.append(DriverRejectedRide(ride_id=self.id, driver_id=old_driver))
    
    def driver_arrived(self):
        self._transition(RideStatus.DRIVER_ARRIVED)
        self._events.append(DriverArrived(ride_id=self.id))
    
    def start_trip(self):
        self._transition(RideStatus.IN_PROGRESS)
        self._started_at = datetime.now()
        self._events.append(TripStarted(ride_id=self.id))
    
    def complete_trip(self, fare: Fare):
        self._transition(RideStatus.COMPLETED)
        self._fare = fare
        self._completed_at = datetime.now()
        self._events.append(TripCompleted(
            ride_id=self.id, driver_id=self._driver_id,
            rider_id=self.rider_id, fare=fare
        ))
    
    def cancel(self, by: str, reason: str):
        self._transition(RideStatus.CANCELLED)
        self._cancellation_reason = reason
        self._events.append(RideCancelled(
            ride_id=self.id, cancelled_by=by, reason=reason
        ))
    
    def assignment_timed_out(self) -> bool:
        if not self._assignment_expires_at:
            return False
        return datetime.now() > self._assignment_expires_at
    
    def _transition(self, new_status: RideStatus):
        allowed = VALID_RIDE_TRANSITIONS.get(self._status, set())
        if new_status not in allowed:
            raise InvalidStateTransition(
                f"Ride cannot transition from {self._status} to {new_status}"
            )
        self._status = new_status
    
    def get_events(self): return list(self._events)
    def clear_events(self): self._events.clear()


# matching.py — Strategy pattern for driver selection

class MatchingStrategy(ABC):
    @abstractmethod
    def find_driver(self, ride: Ride, available_drivers: List["Driver"]) -> Optional["Driver"]:
        pass

class NearestDriverStrategy(MatchingStrategy):
    """Match the closest available driver."""
    def find_driver(self, ride, available_drivers):
        compatible = [d for d in available_drivers
                      if d.vehicle_type == ride.ride_type and d.status == DriverStatus.AVAILABLE]
        if not compatible:
            return None
        return min(compatible, key=lambda d: d.location.distance_km(ride.pickup))

class HighestRatingStrategy(MatchingStrategy):
    """Prefer highest-rated driver within radius."""
    def find_driver(self, ride, available_drivers):
        RADIUS_KM = 5.0
        compatible = [
            d for d in available_drivers
            if d.vehicle_type == ride.ride_type
            and d.status == DriverStatus.AVAILABLE
            and d.location.distance_km(ride.pickup) <= RADIUS_KM
        ]
        if not compatible:
            return None
        return max(compatible, key=lambda d: d.rating)

class EtaBasedStrategy(MatchingStrategy):
    """Match driver with lowest ETA (considers traffic via routing API)."""
    def __init__(self, routing_service):
        self._routing = routing_service
    
    def find_driver(self, ride, available_drivers):
        compatible = [d for d in available_drivers
                      if d.vehicle_type == ride.ride_type and d.status == DriverStatus.AVAILABLE]
        if not compatible:
            return None
        
        etas = [(d, self._routing.get_eta(d.location, ride.pickup))
                for d in compatible[:10]]  # limit API calls to top 10 nearest
        return min(etas, key=lambda x: x[1])[0]


# pricing.py — Strategy for fare calculation
class FareCalculator(ABC):
    @abstractmethod
    def calculate(self, ride: Ride, distance_km: float,
                  duration_minutes: float, surge: Decimal) -> Fare: pass

class StandardFareCalculator(FareCalculator):
    RATES = {
        RideType.ECONOMY: {"base": Decimal("30"), "per_km": Decimal("10"), "per_min": Decimal("1.5")},
        RideType.PREMIUM: {"base": Decimal("50"), "per_km": Decimal("18"), "per_min": Decimal("2.5")},
        RideType.XL:      {"base": Decimal("60"), "per_km": Decimal("15"), "per_min": Decimal("2.0")},
        RideType.AUTO:    {"base": Decimal("20"), "per_km": Decimal("8"),  "per_min": Decimal("1.0")},
    }
    
    def calculate(self, ride, distance_km, duration_minutes, surge) -> Fare:
        rates = self.RATES[ride.ride_type]
        return Fare(
            base_fare=rates["base"],
            distance_fare=Decimal(str(distance_km)) * rates["per_km"],
            time_fare=Decimal(str(duration_minutes)) * rates["per_min"],
            surge_multiplier=surge
        )


# surge.py — Strategy for surge pricing
class SurgeCalculator(ABC):
    @abstractmethod
    def get_surge_multiplier(self, area: Location) -> Decimal: pass

class SupplyDemandSurge(SurgeCalculator):
    def get_surge_multiplier(self, area: Location) -> Decimal:
        nearby_requests = self._count_pending_requests(area)
        available_drivers = self._count_available_drivers(area)
        
        if available_drivers == 0:
            return Decimal("3.0")
        
        ratio = nearby_requests / available_drivers
        if ratio <= 1.0: return Decimal("1.0")
        if ratio <= 1.5: return Decimal("1.2")
        if ratio <= 2.0: return Decimal("1.5")
        if ratio <= 3.0: return Decimal("2.0")
        return Decimal("2.5")


# dispatch_service.py — Orchestrator
class RideDispatchService:
    def __init__(self, ride_repo, driver_repo, matching_strategy: MatchingStrategy,
                 fare_calculator: FareCalculator, surge_calculator: SurgeCalculator,
                 event_bus, routing_service):
        self._rides = ride_repo
        self._drivers = driver_repo
        self._matching = matching_strategy
        self._fare_calc = fare_calculator
        self._surge_calc = surge_calculator
        self._bus = event_bus
        self._routing = routing_service
    
    def request_ride(self, rider_id: UUID, pickup: Location,
                     dropoff: Location, ride_type: RideType) -> Ride:
        ride = Ride(
            ride_id=uuid4(), rider_id=rider_id,
            pickup=pickup, dropoff=dropoff, ride_type=ride_type
        )
        self._rides.save(ride)
        self._dispatch_matching(ride)
        return ride
    
    def _dispatch_matching(self, ride: Ride):
        available_drivers = self._drivers.find_available(ride.ride_type)
        driver = self._matching.find_driver(ride, available_drivers)
        
        if not driver:
            self._bus.publish(NoDriverAvailable(ride_id=ride.id))
            return
        
        ride.assign_driver(driver.id)
        driver.assign_to_ride(ride.id)
        
        self._rides.save(ride)
        self._drivers.save(driver)
        
        for event in ride.get_events():
            self._bus.publish(event)
        ride.clear_events()
    
    def accept_ride(self, driver_id: UUID, ride_id: UUID):
        ride = self._rides.get_or_raise(ride_id)
        if ride.driver_id != driver_id:
            raise PermissionDenied("Driver not assigned to this ride")
        ride.driver_accepted()
        self._rides.save(ride)
        for event in ride.get_events():
            self._bus.publish(event)
        ride.clear_events()
    
    def reject_ride(self, driver_id: UUID, ride_id: UUID):
        ride = self._rides.get_or_raise(ride_id)
        driver = self._drivers.get_or_raise(driver_id)
        
        ride.driver_rejected()
        driver.make_available()  # driver goes back to available pool
        
        self._rides.save(ride)
        self._drivers.save(driver)
        
        for event in ride.get_events():
            self._bus.publish(event)
        ride.clear_events()
        
        # Re-match with different driver
        self._dispatch_matching(ride)
    
    def complete_ride(self, driver_id: UUID, ride_id: UUID):
        ride = self._rides.get_or_raise(ride_id)
        driver = self._drivers.get_or_raise(driver_id)
        
        distance_km = self._routing.calculate_trip_distance(ride)
        duration_minutes = (datetime.now() - ride._started_at).total_seconds() / 60
        surge = self._surge_calc.get_surge_multiplier(ride.pickup)
        fare = self._fare_calc.calculate(ride, distance_km, duration_minutes, surge)
        
        ride.complete_trip(fare)
        driver.complete_trip()  # driver becomes available again
        
        self._rides.save(ride)
        self._drivers.save(driver)
        
        for event in ride.get_events():
            self._bus.publish(event)
        ride.clear_events()


# Background job: handle assignment timeouts
class AssignmentTimeoutHandler:
    """Periodically checks for expired assignments and re-dispatches."""
    
    def run(self, dispatch_service: RideDispatchService, ride_repo):
        assigned_rides = ride_repo.find_by_status(RideStatus.DRIVER_ASSIGNED)
        for ride in assigned_rides:
            if ride.assignment_timed_out():
                driver = driver_repo.get(ride.driver_id)
                driver.make_available()
                driver_repo.save(driver)
                ride.driver_rejected()  # triggers re-matching event
                ride_repo.save(ride)
                dispatch_service._dispatch_matching(ride)
```

## Step 4: Patterns Used

|Pattern|Where|Why|
|---|---|---|
|**State**|`Ride._status` + `Driver.status`|Complex lifecycle with many transitions|
|**Strategy**|`MatchingStrategy`, `FareCalculator`, `SurgeCalculator`|All three vary independently and are swappable|
|**Observer**|Domain events via `event_bus`|Payment, notifications, analytics decoupled from dispatch|
|**Command**|`RideDispatchService` methods as commands|Each action is encapsulated, retryable|
