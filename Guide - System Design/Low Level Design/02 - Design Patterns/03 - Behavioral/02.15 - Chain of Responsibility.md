# Chain of Responsibility Pattern

## At a Glance

|||
|---|---|
|**Type**|Behavioral|
|**Intent**|Pass a request along a chain of handlers; each handler decides to process it or pass it on|
|**Also Known As**|Middleware Pipeline|
|**Frequency in Backend**|★★★★★ — middleware stacks, validation pipelines, auth chains|

---

## The Problem It Solves

A request needs multiple independent processing steps. Without this pattern, one monolithic handler grows bloated, and adding/removing steps requires modifying it.

---

## Full Backend Example: HTTP Middleware Pipeline

```python
# middleware.py
from abc import ABC, abstractmethod
from dataclasses import dataclass
from typing import Optional, Callable

@dataclass
class HttpRequest:
    method: str
    path: str
    headers: dict
    body: bytes
    query_params: dict
    user: Optional["AuthenticatedUser"] = None
    attributes: dict = None  # for passing data between handlers
    
    def __post_init__(self):
        self.attributes = self.attributes or {}

@dataclass
class HttpResponse:
    status_code: int
    body: bytes
    headers: dict
    
    @classmethod
    def ok(cls, body: bytes, headers: dict = None):
        return cls(200, body, headers or {"Content-Type": "application/json"})
    
    @classmethod
    def error(cls, status: int, message: str):
        import json
        return cls(status, json.dumps({"error": message}).encode(), 
                  {"Content-Type": "application/json"})


# Handler interface
class Middleware(ABC):
    def __init__(self):
        self._next: Optional["Middleware"] = None
    
    def set_next(self, handler: "Middleware") -> "Middleware":
        self._next = handler
        return handler  # enables chaining: a.set_next(b).set_next(c)
    
    @abstractmethod
    def handle(self, request: HttpRequest) -> HttpResponse: pass
    
    def _proceed(self, request: HttpRequest) -> HttpResponse:
        if self._next:
            return self._next.handle(request)
        return HttpResponse.error(404, "No handler found")


# Concrete middleware handlers
class RateLimitMiddleware(Middleware):
    """Blocks requests exceeding rate limit."""
    
    def __init__(self, requests_per_minute: int, rate_limiter):
        super().__init__()
        self._rpm = requests_per_minute
        self._limiter = rate_limiter
    
    def handle(self, request: HttpRequest) -> HttpResponse:
        ip = request.headers.get("X-Forwarded-For", "unknown")
        
        if not self._limiter.allow(ip, self._rpm):
            return HttpResponse.error(429, "Rate limit exceeded. Try again later.")
        
        return self._proceed(request)   # pass to next


class CorsMiddleware(Middleware):
    """Handles CORS preflight and adds CORS headers."""
    
    def __init__(self, allowed_origins: list[str]):
        super().__init__()
        self._allowed_origins = allowed_origins
    
    def handle(self, request: HttpRequest) -> HttpResponse:
        origin = request.headers.get("Origin", "")
        
        if request.method == "OPTIONS":   # preflight — short-circuit
            return HttpResponse(
                status_code=204,
                body=b"",
                headers={
                    "Access-Control-Allow-Origin": origin if origin in self._allowed_origins else "",
                    "Access-Control-Allow-Methods": "GET, POST, PUT, DELETE",
                    "Access-Control-Allow-Headers": "Content-Type, Authorization",
                }
            )
        
        response = self._proceed(request)
        
        # Add CORS headers to every response
        if origin in self._allowed_origins:
            response.headers["Access-Control-Allow-Origin"] = origin
        
        return response


class AuthMiddleware(Middleware):
    """Validates JWT token, attaches user to request."""
    
    PUBLIC_PATHS = {"/health", "/login", "/register", "/docs"}
    
    def __init__(self, jwt_service):
        super().__init__()
        self._jwt = jwt_service
    
    def handle(self, request: HttpRequest) -> HttpResponse:
        if request.path in self.PUBLIC_PATHS:
            return self._proceed(request)   # skip auth for public routes
        
        auth_header = request.headers.get("Authorization", "")
        if not auth_header.startswith("Bearer "):
            return HttpResponse.error(401, "Missing or invalid authorization header")
        
        token = auth_header[7:]
        try:
            user = self._jwt.verify(token)
            request.user = user   # attach to request for downstream handlers
        except TokenExpiredError:
            return HttpResponse.error(401, "Token expired")
        except InvalidTokenError:
            return HttpResponse.error(401, "Invalid token")
        
        return self._proceed(request)


class RequestLoggingMiddleware(Middleware):
    """Logs request/response for observability."""
    
    def handle(self, request: HttpRequest) -> HttpResponse:
        import time
        start = time.monotonic()
        
        response = self._proceed(request)
        
        elapsed_ms = (time.monotonic() - start) * 1000
        logger.info(
            f"{request.method} {request.path} → {response.status_code} "
            f"[{elapsed_ms:.1f}ms] user={getattr(request.user, 'id', 'anonymous')}"
        )
        return response


class RequestValidationMiddleware(Middleware):
    """Validates Content-Type, body size limits."""
    
    MAX_BODY_SIZE = 10 * 1024 * 1024  # 10MB
    
    def handle(self, request: HttpRequest) -> HttpResponse:
        if request.method in ("POST", "PUT", "PATCH"):
            if len(request.body) > self.MAX_BODY_SIZE:
                return HttpResponse.error(413, "Request body too large")
            
            content_type = request.headers.get("Content-Type", "")
            if "application/json" not in content_type:
                return HttpResponse.error(415, "Content-Type must be application/json")
        
        return self._proceed(request)


# pipeline.py — Build and run the chain
class MiddlewarePipeline:
    def __init__(self, middlewares: list[Middleware]):
        if not middlewares:
            raise ValueError("Pipeline must have at least one handler")
        
        # Chain them: first → second → ... → last
        for i in range(len(middlewares) - 1):
            middlewares[i].set_next(middlewares[i + 1])
        
        self._head = middlewares[0]
    
    def process(self, request: HttpRequest) -> HttpResponse:
        return self._head.handle(request)


# Wiring — order matters!
pipeline = MiddlewarePipeline([
    RequestLoggingMiddleware(),            # 1. Always log
    CorsMiddleware(["https://app.com"]),   # 2. CORS (handles OPTIONS before auth)
    RateLimitMiddleware(100, redis_limiter), # 3. Rate limit
    RequestValidationMiddleware(),         # 4. Validate format
    AuthMiddleware(jwt_service),           # 5. Auth (after format validation)
    RouteHandlerMiddleware(router),        # 6. Business logic
])

response = pipeline.process(incoming_request)
```

## Validation Pipeline (Another Classic Use)

```python
class OrderValidator(ABC):
    def set_next(self, v: "OrderValidator") -> "OrderValidator":
        self._next = v; return v
    
    def validate(self, order) -> list[str]:
        errors = self._check(order)
        if self._next:
            errors.extend(self._next.validate(order))
        return errors
    
    @abstractmethod
    def _check(self, order) -> list[str]: pass

class ItemsValidator(OrderValidator):
    def _check(self, order):
        return ["Order must have items"] if not order.items else []

class TotalValidator(OrderValidator):
    def _check(self, order):
        return ["Total must be positive"] if order.total <= 0 else []

class InventoryValidator(OrderValidator):
    def _check(self, order):
        return [f"{item.sku} out of stock" 
                for item in order.items 
                if not inventory.has(item.sku, item.qty)]

# Chain
v = ItemsValidator()
v.set_next(TotalValidator()).set_next(InventoryValidator())
errors = v.validate(order)
```

## When NOT to Use Chain of Responsibility

- **All handlers must process** — just iterate a list and call each directly
- **Handler order can't be wrong** — explicit sequential calls are clearer
- **Single handler always** — no chain needed

---

## Tradeoffs

|Pro|Con|
|---|---|
|Add/remove/reorder handlers easily|Request may fall through with no handler (need fallback)|
|Single Responsibility per handler|No guarantee of processing|
|Testable in isolation|Debug is harder — trace through chain|
|Open/Closed — new handler = new class||
