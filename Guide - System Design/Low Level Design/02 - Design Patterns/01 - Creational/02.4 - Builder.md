# Builder Pattern

## At a Glance

|||
|---|---|
|**Type**|Creational|
|**Intent**|Construct complex objects step-by-step, separating construction from representation|
|**Frequency in Backend**|★★★★☆ — query builders, HTTP clients, config, test fixtures|

---

## The Problem It Solves

Telescoping constructors — when a class needs many optional parameters, you end up with unreadable multi-argument constructors or too many overloads.

```python
# ❌ Telescoping constructor — unreadable, error-prone
query = SqlQuery("orders", ["id","total"], "status='PAID'", 
                 "created_at DESC", 20, 0, True, ["items"], None, "user_id=123")
# What does the 7th argument mean? No IDE help, easy to swap args.
```

## Full Backend Example: SQL Query Builder

```python
# query_builder.py
from dataclasses import dataclass, field
from typing import Optional, List, Any
from enum import Enum

class SortDirection(Enum):
    ASC = "ASC"
    DESC = "DESC"

@dataclass
class JoinClause:
    table: str
    condition: str
    join_type: str = "INNER"

@dataclass
class SqlQuery:
    """Immutable query — only Builder can create it."""
    table: str
    columns: List[str]
    conditions: List[str]
    params: List[Any]
    order_by: Optional[str]
    sort_direction: SortDirection
    limit: Optional[int]
    offset: int
    joins: List[JoinClause]
    
    def to_sql(self) -> tuple[str, list]:
        """Render to SQL string + parameter list."""
        select = ", ".join(self.columns) if self.columns else "*"
        sql = f"SELECT {select} FROM {self.table}"
        
        for join in self.joins:
            sql += f" {join.join_type} JOIN {join.table} ON {join.condition}"
        
        if self.conditions:
            sql += " WHERE " + " AND ".join(self.conditions)
        
        if self.order_by:
            sql += f" ORDER BY {self.order_by} {self.sort_direction.value}"
        
        if self.limit is not None:
            sql += f" LIMIT {self.limit}"
        
        if self.offset:
            sql += f" OFFSET {self.offset}"
        
        return sql, self.params


class QueryBuilder:
    """
    Fluent builder for SqlQuery.
    Each method returns self — enables method chaining.
    """
    
    def __init__(self, table: str):
        self._table = table
        self._columns: List[str] = []
        self._conditions: List[str] = []
        self._params: List[Any] = []
        self._order_by: Optional[str] = None
        self._sort_direction = SortDirection.ASC
        self._limit: Optional[int] = None
        self._offset: int = 0
        self._joins: List[JoinClause] = []
    
    def select(self, *columns: str) -> "QueryBuilder":
        self._columns.extend(columns)
        return self
    
    def where(self, condition: str, *params) -> "QueryBuilder":
        self._conditions.append(condition)
        self._params.extend(params)
        return self
    
    def where_eq(self, column: str, value) -> "QueryBuilder":
        return self.where(f"{column} = %s", value)
    
    def where_in(self, column: str, values: list) -> "QueryBuilder":
        placeholders = ", ".join(["%s"] * len(values))
        return self.where(f"{column} IN ({placeholders})", *values)
    
    def where_like(self, column: str, pattern: str) -> "QueryBuilder":
        return self.where(f"{column} LIKE %s", pattern)
    
    def order_by(self, column: str, direction: SortDirection = SortDirection.ASC) -> "QueryBuilder":
        self._order_by = column
        self._sort_direction = direction
        return self
    
    def limit(self, n: int) -> "QueryBuilder":
        self._limit = n
        return self
    
    def offset(self, n: int) -> "QueryBuilder":
        self._offset = n
        return self
    
    def join(self, table: str, condition: str,
             join_type: str = "INNER") -> "QueryBuilder":
        self._joins.append(JoinClause(table, condition, join_type))
        return self
    
    def left_join(self, table: str, condition: str) -> "QueryBuilder":
        return self.join(table, condition, "LEFT")
    
    def paginate(self, page: int, per_page: int) -> "QueryBuilder":
        self._limit = per_page
        self._offset = (page - 1) * per_page
        return self
    
    def build(self) -> SqlQuery:
        if not self._table:
            raise ValueError("Table is required")
        return SqlQuery(
            table=self._table,
            columns=list(self._columns),
            conditions=list(self._conditions),
            params=list(self._params),
            order_by=self._order_by,
            sort_direction=self._sort_direction,
            limit=self._limit,
            offset=self._offset,
            joins=list(self._joins)
        )


# Usage — readable, self-documenting
query = (QueryBuilder("orders")
    .select("orders.id", "orders.total", "users.email")
    .left_join("users", "orders.user_id = users.id")
    .where_eq("orders.status", "PAID")
    .where("orders.total > %s", 1000)
    .where_in("orders.currency", ["INR", "USD"])
    .order_by("orders.created_at", SortDirection.DESC)
    .paginate(page=2, per_page=20)
    .build()
)

sql, params = query.to_sql()
# SELECT orders.id, orders.total, users.email
# FROM orders
# LEFT JOIN users ON orders.user_id = users.id
# WHERE orders.status = %s AND orders.total > %s AND orders.currency IN (%s, %s)
# ORDER BY orders.created_at DESC
# LIMIT 20 OFFSET 20


# http_request_builder.py — another classic Builder use case
@dataclass
class HttpRequest:
    method: str
    url: str
    headers: dict
    query_params: dict
    body: Optional[bytes]
    timeout_seconds: int
    retry_count: int
    follow_redirects: bool

class HttpRequestBuilder:
    def __init__(self, method: str, url: str):
        self._method = method.upper()
        self._url = url
        self._headers: dict = {}
        self._query_params: dict = {}
        self._body: Optional[bytes] = None
        self._timeout = 30
        self._retries = 0
        self._follow_redirects = True
    
    def header(self, key: str, value: str) -> "HttpRequestBuilder":
        self._headers[key] = value
        return self
    
    def bearer_token(self, token: str) -> "HttpRequestBuilder":
        return self.header("Authorization", f"Bearer {token}")
    
    def content_type(self, ct: str) -> "HttpRequestBuilder":
        return self.header("Content-Type", ct)
    
    def json_body(self, data: dict) -> "HttpRequestBuilder":
        import json
        self._body = json.dumps(data).encode()
        return self.content_type("application/json")
    
    def param(self, key: str, value: str) -> "HttpRequestBuilder":
        self._query_params[key] = value
        return self
    
    def timeout(self, seconds: int) -> "HttpRequestBuilder":
        self._timeout = seconds
        return self
    
    def retries(self, n: int) -> "HttpRequestBuilder":
        self._retries = n
        return self
    
    def build(self) -> HttpRequest:
        return HttpRequest(
            method=self._method,
            url=self._url,
            headers=dict(self._headers),
            query_params=dict(self._query_params),
            body=self._body,
            timeout_seconds=self._timeout,
            retry_count=self._retries,
            follow_redirects=self._follow_redirects
        )

# Clean, readable construction
request = (HttpRequestBuilder("POST", "https://api.payments.com/charge")
    .bearer_token(token)
    .json_body({"amount": 999, "currency": "INR"})
    .timeout(10)
    .retries(3)
    .build()
)
```

## Test Data Builder (Vital for Testing)

```python
# test_builders.py — Builder for creating test fixtures
class OrderBuilder:
    """Fluent builder for test Order objects with sensible defaults."""
    
    def __init__(self):
        self._id = uuid4()
        self._user_id = uuid4()
        self._status = OrderStatus.DRAFT
        self._items = []
        self._total = Decimal("100.00")
        self._created_at = datetime.now()
    
    def with_id(self, id): self._id = id; return self
    def with_user(self, user_id): self._user_id = user_id; return self
    def as_paid(self): self._status = OrderStatus.PAID; return self
    def as_cancelled(self): self._status = OrderStatus.CANCELLED; return self
    def with_total(self, amount): self._total = Decimal(str(amount)); return self
    def with_item(self, product_id, qty=1, price=50):
        self._items.append(OrderItem(product_id=product_id, qty=qty, price=Decimal(str(price))))
        return self
    
    def build(self) -> Order:
        return Order(id=self._id, user_id=self._user_id,
                    status=self._status, items=self._items,
                    total=self._total, created_at=self._created_at)

# In tests:
def test_cancel_paid_order_fails():
    order = OrderBuilder().as_paid().with_total(500).build()
    with pytest.raises(InvalidStateTransition):
        order.cancel("Changed mind")
```

## When NOT to Use Builder

- **Few parameters** (< 3-4, all required) — just use a constructor
- **No optional parameters** — no value in a builder
- **Immutable simple objects** — `@dataclass(frozen=True)` is cleaner

---

## Tradeoffs

|Pro|Con|
|---|---|
|Readable, self-documenting|More code than a constructor|
|Optional parameters handled cleanly|Builder itself needs testing|
|Step-by-step construction|Mutable intermediate state (until `build()`)|
|Test fixture creation|Can create partially-built invalid objects if `build()` not validated|

---

## Common Mistakes

1. **Forgetting validation in `build()`** — builder should validate before returning
2. **Mutable product** — if the built object is mutable, callers can corrupt it after building
3. **Too many methods** — if builder has 30 methods, the object is probably too complex
