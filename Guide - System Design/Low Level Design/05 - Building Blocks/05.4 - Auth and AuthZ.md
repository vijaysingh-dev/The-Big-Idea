# Auth and AuthZ

## AuthN vs AuthZ

```
Authentication (AuthN):   WHO are you?         → Identity verification
Authorization (AuthZ):    WHAT can you do?      → Permission check

AuthN happens first. AuthZ uses the verified identity.
```

## JWT (JSON Web Token)

### Structure

```
Header.Payload.Signature

Header:  {"alg": "HS256", "typ": "JWT"}
Payload: {"sub": "user-123", "role": "admin", "exp": 1700000000, "iat": 1699990000}
Signature: HMACSHA256(base64(header) + "." + base64(payload), secret)
```

### Implementation

```python
# jwt_service.py
import jwt
from datetime import datetime, timedelta
from typing import Optional

class JwtService:
    ACCESS_TOKEN_TTL = timedelta(minutes=15)
    REFRESH_TOKEN_TTL = timedelta(days=7)
    
    def __init__(self, secret_key: str, algorithm: str = "HS256"):
        self._secret = secret_key
        self._algo = algorithm
    
    def generate_access_token(self, user_id: str, roles: list[str]) -> str:
        payload = {
            "sub": user_id,
            "roles": roles,
            "type": "access",
            "iat": datetime.utcnow(),
            "exp": datetime.utcnow() + self.ACCESS_TOKEN_TTL,
        }
        return jwt.encode(payload, self._secret, algorithm=self._algo)
    
    def generate_refresh_token(self, user_id: str) -> str:
        payload = {
            "sub": user_id,
            "type": "refresh",
            "jti": str(uuid4()),         # unique token ID for revocation
            "iat": datetime.utcnow(),
            "exp": datetime.utcnow() + self.REFRESH_TOKEN_TTL,
        }
        return jwt.encode(payload, self._secret, algorithm=self._algo)
    
    def verify(self, token: str) -> dict:
        try:
            payload = jwt.decode(token, self._secret, algorithms=[self._algo])
            return payload
        except jwt.ExpiredSignatureError:
            raise TokenExpiredError("Token has expired")
        except jwt.InvalidTokenError as e:
            raise InvalidTokenError(f"Invalid token: {e}")
    
    def refresh(self, refresh_token: str) -> tuple[str, str]:
        payload = self.verify(refresh_token)
        
        if payload.get("type") != "refresh":
            raise InvalidTokenError("Not a refresh token")
        
        user_id = payload["sub"]
        # Check if refresh token has been revoked
        if self._is_revoked(payload["jti"]):
            raise TokenRevokedError("Refresh token has been revoked")
        
        # Generate new pair
        return (
            self.generate_access_token(user_id, self._get_user_roles(user_id)),
            self.generate_refresh_token(user_id)
        )
    
    def revoke(self, jti: str):
        """Add token JTI to revocation list (Redis with TTL)."""
        self._revocation_store.add(jti, ttl=int(self.REFRESH_TOKEN_TTL.total_seconds()))
```

### JWT Tradeoffs

```
✅ Stateless — no server-side session storage
✅ Works across services — any service can verify with shared secret
✅ Carries claims — roles/permissions embedded

❌ Cannot revoke before expiry (without a token store/blacklist)
❌ Payload grows with more claims — sent with every request
❌ Secret rotation invalidates all tokens

Mitigation:
- Short access token TTL (15min) — limited damage if stolen
- Refresh token in httpOnly cookie — not accessible to JS
- Refresh token rotation — issue new refresh on each use
- JTI blacklist for logout (Redis, TTL = refresh token TTL)
```

## RBAC (Role-Based Access Control)

```python
# rbac.py
from enum import Enum
from dataclasses import dataclass

class Permission(Enum):
    # Orders
    ORDER_READ = "order:read"
    ORDER_WRITE = "order:write"
    ORDER_CANCEL = "order:cancel"
    ORDER_ADMIN = "order:admin"
    
    # Users
    USER_READ = "user:read"
    USER_WRITE = "user:write"
    USER_ADMIN = "user:admin"
    
    # Products
    PRODUCT_READ = "product:read"
    PRODUCT_WRITE = "product:write"

ROLE_PERMISSIONS = {
    "customer": {
        Permission.ORDER_READ, Permission.ORDER_WRITE, Permission.ORDER_CANCEL,
        Permission.PRODUCT_READ, Permission.USER_READ
    },
    "support": {
        Permission.ORDER_READ, Permission.ORDER_ADMIN,
        Permission.USER_READ, Permission.PRODUCT_READ
    },
    "catalog_manager": {
        Permission.PRODUCT_READ, Permission.PRODUCT_WRITE,
        Permission.ORDER_READ
    },
    "admin": set(Permission),  # all permissions
}

class RbacService:
    def has_permission(self, user_roles: list[str], permission: Permission) -> bool:
        for role in user_roles:
            if permission in ROLE_PERMISSIONS.get(role, set()):
                return True
        return False
    
    def assert_permission(self, user_roles: list[str], permission: Permission):
        if not self.has_permission(user_roles, permission):
            raise PermissionDenied(f"Permission required: {permission.value}")


# Decorator for route-level auth
def require_permission(permission: Permission):
    def decorator(func):
        @functools.wraps(func)
        def wrapper(*args, request: HttpRequest, **kwargs):
            rbac.assert_permission(request.user.roles, permission)
            return func(*args, request=request, **kwargs)
        return wrapper
    return decorator

@require_permission(Permission.ORDER_ADMIN)
def cancel_any_order(request, order_id):
    ...
```

## ABAC (Attribute-Based Access Control)

RBAC: "can user with role X do action Y?" ABAC: "can user with attributes A,B,C do action Y on resource with attributes D,E,F?"

```python
# ABAC policy
class OrderAccessPolicy:
    def can_view(self, user: User, order: Order) -> bool:
        # Admin can view any order
        if "admin" in user.roles:
            return True
        
        # Support can view any order in their assigned region
        if "support" in user.roles and order.region == user.region:
            return True
        
        # Customer can only view their own orders
        return order.user_id == user.id
    
    def can_cancel(self, user: User, order: Order) -> bool:
        if not self.can_view(user, order):
            return False
        
        # Can only cancel PENDING orders
        if order.status not in (OrderStatus.PENDING, OrderStatus.CONFIRMED):
            return False
        
        # Customer can only cancel within 30 minutes
        if "admin" not in user.roles:
            age_minutes = (datetime.now() - order.created_at).total_seconds() / 60
            if age_minutes > 30:
                return False
        
        return True


# Service uses policy
class OrderService:
    def __init__(self, repo, policy: OrderAccessPolicy):
        self._repo = repo
        self._policy = policy
    
    def cancel_order(self, user: User, order_id: UUID, reason: str):
        order = self._repo.get_or_raise(order_id)
        
        if not self._policy.can_cancel(user, order):
            raise PermissionDenied(f"Cannot cancel order {order_id}")
        
        order.cancel(reason)
        self._repo.save(order)
```

## API Key Authentication

```python
class ApiKeyService:
    def __init__(self, api_key_repo):
        self._repo = api_key_repo
    
    def generate(self, client_id: str, permissions: list[str]) -> str:
        raw_key = secrets.token_urlsafe(32)   # 256 bits of entropy
        hashed = hashlib.sha256(raw_key.encode()).hexdigest()
        
        self._repo.save(ApiKey(
            key_hash=hashed,          # never store raw key
            client_id=client_id,
            permissions=permissions,
            created_at=datetime.now(),
            last_used_at=None
        ))
        
        return raw_key  # only time the raw key is shown
    
    def verify(self, raw_key: str) -> ApiKey:
        hashed = hashlib.sha256(raw_key.encode()).hexdigest()
        api_key = self._repo.find_by_hash(hashed)
        
        if not api_key:
            raise InvalidApiKeyError()
        
        if api_key.revoked:
            raise RevokedApiKeyError()
        
        # Update last used (async, non-blocking in production)
        api_key.last_used_at = datetime.now()
        self._repo.save(api_key)
        
        return api_key
```

## Common Auth Pitfalls

|Pitfall|Fix|
|---|---|
|Storing plaintext passwords|Always bcrypt/argon2 hash|
|Logging tokens or secrets|Scrub from logs|
|Long-lived access tokens|TTL 15min, use refresh tokens|
|JWT without expiry|Always set `exp`|
|Trusting user-supplied role claims|Verify roles server-side, not from token payload alone|
|HTTP for token transmission|HTTPS always|
|Missing rate limiting on login|Brute force protection|
