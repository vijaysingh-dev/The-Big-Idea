# Build Your Own: Job Scheduler

> **Revision tags:** #build-your-own #scheduling  
> **Related:** [[03.15 - Schedulers and Cron]]

---

## Goal

Build a cron-style job scheduler that:
- Registers jobs with cron expressions
- Fires jobs at the correct time
- Prevents concurrent execution of the same job (if configured)
- Tracks run history and status
- Handles missed runs on restart

---

## Core Design

```
[Job Registry]     → stores job definitions + cron schedules
[Scheduler Loop]   → every second: check which jobs are due → enqueue them
[Worker Pool]      → picks up enqueued jobs → executes → records result
[Run History]      → append-only log of each execution (start, end, status)
[Concurrency Lock] → per-job lock: skip if previous run still active
```

## Implementation

```python
import threading
import time
import uuid
import logging
from dataclasses import dataclass, field
from typing import Callable, Dict, List, Optional
from datetime import datetime, timezone
from enum import Enum
from queue import Queue

logger = logging.getLogger(__name__)

# ── Cron expression parser (simplified: minute hour day month weekday) ────────

def cron_is_due(cron_expr: str, dt: datetime) -> bool:
    """
    Check if a cron expression fires at the given datetime.
    Supports: * (any), */N (every N), single values.
    Format: "minute hour day month weekday"
    Examples:
        "0 * * * *"   → every hour at :00
        "*/5 * * * *" → every 5 minutes
        "0 2 * * *"   → 2:00 AM daily
        "0 9 * * 1"   → Monday 9 AM (weekday: 0=Sun, 1=Mon...)
    """
    parts = cron_expr.strip().split()
    if len(parts) != 5:
        raise ValueError(f"Invalid cron expression: {cron_expr}")

    minute, hour, day, month, weekday = parts
    checks = [
        (minute,  dt.minute),
        (hour,    dt.hour),
        (day,     dt.day),
        (month,   dt.month),
        (weekday, dt.weekday() + 1),  # Monday=1 in our convention
    ]

    for expr, value in checks:
        if expr == "*":
            continue
        if expr.startswith("*/"):
            step = int(expr[2:])
            if value % step != 0:
                return False
        elif int(expr) != value:
            return False
    return True


# ── Job definition ────────────────────────────────────────────────────────────

class JobStatus(Enum):
    PENDING   = "pending"
    RUNNING   = "running"
    SUCCESS   = "success"
    FAILED    = "failed"
    SKIPPED   = "skipped"   # skipped due to concurrency lock


@dataclass
class JobRun:
    run_id: str
    job_name: str
    status: JobStatus
    started_at: Optional[float] = None
    finished_at: Optional[float] = None
    error: Optional[str] = None

    @property
    def duration_seconds(self) -> Optional[float]:
        if self.started_at and self.finished_at:
            return self.finished_at - self.started_at
        return None


@dataclass
class Job:
    name: str
    func: Callable
    schedule: str            # cron expression
    allow_concurrent: bool = False   # if False, skip if previous run still active
    max_retries: int = 0
    timeout_seconds: Optional[float] = None

    # Internal state
    _running: bool = field(default=False, init=False, repr=False)
    _lock: threading.Lock = field(default_factory=threading.Lock, init=False, repr=False)

    def is_running(self) -> bool:
        with self._lock:
            return self._running

    def set_running(self, value: bool):
        with self._lock:
            self._running = value


# ── Scheduler ─────────────────────────────────────────────────────────────────

class JobScheduler:
    """
    Cron-style job scheduler.
    Runs a tick loop every second, dispatches due jobs to a worker pool.
    """

    def __init__(self, worker_count: int = 4):
        self._jobs: Dict[str, Job] = {}
        self._run_history: List[JobRun] = []
        self._history_lock = threading.Lock()
        self._job_queue: Queue = Queue()
        self._running = False

        # Worker pool
        self._workers = [
            threading.Thread(target=self._worker_loop, daemon=True)
            for _ in range(worker_count)
        ]

        # Scheduler tick thread
        self._tick_thread = threading.Thread(target=self._tick_loop, daemon=True)

    # ── Public API ────────────────────────────────────────────────────────────

    def register(self, name: str, func: Callable, schedule: str,
                 allow_concurrent: bool = False,
                 max_retries: int = 0,
                 timeout_seconds: Optional[float] = None) -> Job:
        """Register a job with the scheduler."""
        job = Job(
            name=name,
            func=func,
            schedule=schedule,
            allow_concurrent=allow_concurrent,
            max_retries=max_retries,
            timeout_seconds=timeout_seconds
        )
        self._jobs[name] = job
        logger.info(f"[Scheduler] Registered job '{name}' → schedule: '{schedule}'")
        return job

    def start(self):
        """Start the scheduler and worker pool."""
        self._running = True
        for w in self._workers:
            w.start()
        self._tick_thread.start()
        logger.info(f"[Scheduler] Started with {len(self._workers)} workers")

    def stop(self):
        """Graceful shutdown."""
        self._running = False

    def run_now(self, job_name: str):
        """Manually trigger a job immediately."""
        if job_name not in self._jobs:
            raise ValueError(f"Unknown job: {job_name}")
        self._enqueue_job(self._jobs[job_name])

    def history(self, job_name: Optional[str] = None,
                limit: int = 20) -> List[JobRun]:
        with self._history_lock:
            runs = self._run_history
            if job_name:
                runs = [r for r in runs if r.job_name == job_name]
            return list(reversed(runs))[:limit]

    def stats(self) -> dict:
        with self._history_lock:
            return {
                "registered_jobs": list(self._jobs.keys()),
                "queue_depth": self._job_queue.qsize(),
                "total_runs": len(self._run_history),
                "running_jobs": [j.name for j in self._jobs.values() if j.is_running()]
            }

    # ── Internal ──────────────────────────────────────────────────────────────

    def _tick_loop(self):
        """
        Runs every second. Checks each job's cron schedule against
        the current time and enqueues due jobs.
        """
        while self._running:
            tick_start = time.time()
            now = datetime.now(timezone.utc)

            for job in list(self._jobs.values()):
                try:
                    if cron_is_due(job.schedule, now):
                        self._enqueue_job(job)
                except Exception as e:
                    logger.error(f"[Scheduler] Error checking job '{job.name}': {e}")

            # Sleep for the remainder of the second
            elapsed = time.time() - tick_start
            time.sleep(max(0, 1.0 - elapsed))

    def _enqueue_job(self, job: Job):
        """Put a job on the execution queue, respecting concurrency settings."""
        if not job.allow_concurrent and job.is_running():
            run = JobRun(
                run_id=str(uuid.uuid4()),
                job_name=job.name,
                status=JobStatus.SKIPPED
            )
            self._record_run(run)
            logger.info(f"[Scheduler] Skipping '{job.name}' — previous run still active")
            return

        run = JobRun(
            run_id=str(uuid.uuid4()),
            job_name=job.name,
            status=JobStatus.PENDING
        )
        self._job_queue.put((job, run))

    def _worker_loop(self):
        """Worker thread: picks jobs from the queue and executes them."""
        while self._running:
            try:
                job, run = self._job_queue.get(timeout=1.0)
                self._execute_job(job, run)
            except Exception:
                continue  # queue.get timeout — loop again

    def _execute_job(self, job: Job, run: JobRun):
        """Execute a single job run with retry support."""
        job.set_running(True)
        run.status = JobStatus.RUNNING
        run.started_at = time.time()

        attempt = 0
        last_error = None

        while attempt <= job.max_retries:
            try:
                if job.timeout_seconds:
                    # Run in thread with timeout
                    result_holder = [None]
                    error_holder = [None]

                    def target():
                        try:
                            result_holder[0] = job.func()
                        except Exception as e:
                            error_holder[0] = e

                    t = threading.Thread(target=target, daemon=True)
                    t.start()
                    t.join(timeout=job.timeout_seconds)

                    if t.is_alive():
                        raise TimeoutError(
                            f"Job '{job.name}' timed out after {job.timeout_seconds}s"
                        )
                    if error_holder[0]:
                        raise error_holder[0]
                else:
                    job.func()

                # Success
                run.status = JobStatus.SUCCESS
                run.finished_at = time.time()
                logger.info(
                    f"[Scheduler] '{job.name}' completed in "
                    f"{run.duration_seconds:.2f}s"
                )
                break

            except Exception as e:
                last_error = e
                attempt += 1
                if attempt <= job.max_retries:
                    logger.warning(
                        f"[Scheduler] '{job.name}' failed (attempt {attempt}/"
                        f"{job.max_retries + 1}): {e}. Retrying..."
                    )
                    time.sleep(2 ** attempt)  # simple backoff between retries
                else:
                    run.status = JobStatus.FAILED
                    run.error = str(last_error)
                    run.finished_at = time.time()
                    logger.error(
                        f"[Scheduler] '{job.name}' FAILED after "
                        f"{attempt} attempts: {last_error}"
                    )

        job.set_running(False)
        self._record_run(run)

    def _record_run(self, run: JobRun):
        with self._history_lock:
            self._run_history.append(run)
            # Keep last 1000 runs in memory
            if len(self._run_history) > 1000:
                self._run_history = self._run_history[-1000:]
```

## Usage

```python
import time

scheduler = JobScheduler(worker_count=4)

# Register jobs
def generate_daily_report():
    print("Generating daily report...")
    time.sleep(2)
    print("Report done.")

def sync_exchange_rates():
    print("Syncing exchange rates...")

def cleanup_expired_sessions():
    print("Cleaning up expired sessions...")

scheduler.register("daily_report",
                   generate_daily_report,
                   schedule="0 2 * * *",       # 2 AM daily
                   allow_concurrent=False,
                   timeout_seconds=300)

scheduler.register("sync_rates",
                   sync_exchange_rates,
                   schedule="*/5 * * * *",     # every 5 minutes
                   max_retries=3)

scheduler.register("session_cleanup",
                   cleanup_expired_sessions,
                   schedule="0 * * * *",        # every hour at :00
                   allow_concurrent=False)

scheduler.start()

# Manually trigger for testing
scheduler.run_now("sync_rates")

time.sleep(10)
print(scheduler.stats())
print(scheduler.history("sync_rates"))
```

## What Production Systems Add

| Feature | What's Missing Here |
|---|---|
| Distributed leader election | Multiple scheduler nodes — only one fires each job |
| Persistent job store | Jobs and history survive restarts (Postgres, Redis) |
| Missed run catch-up | On restart, detect and backfill missed scheduled runs |
| Web UI | Airflow/Celery Beat have dashboards for job monitoring |
| DAG dependencies | Job B only runs after Job A completes |
| Pause / resume | Temporarily disable jobs without unregistering |
| Dynamic registration | Add/remove jobs at runtime without restart |
| Distributed worker pool | Workers on multiple machines consuming from shared queue |
| Alerting on failure | PagerDuty / Slack alert when job fails repeatedly |

---

## Key Takeaways

- The tick loop checks every job every second — at production scale, store next-run timestamps to avoid O(jobs) scan per second
- Concurrency control is per-job, not global — a slow job only blocks itself
- Workers are decoupled from the scheduler — the queue is the boundary
- Timeout runs in a thread because Python has no native thread cancellation — the thread may continue running after timeout (use `multiprocessing` for true cancellation)
