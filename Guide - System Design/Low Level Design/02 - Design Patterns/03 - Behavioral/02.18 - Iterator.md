# Iterator Pattern

## At a Glance

|||
|---|---|
|**Type**|Behavioral|
|**Intent**|Provide a way to access elements of a collection sequentially without exposing its underlying structure|
|**Frequency in Backend**|★★★☆☆ — cursor pagination, batch processing, streaming large datasets|

---

## The Problem It Solves

You need to iterate over a large dataset (millions of DB rows, paginated API results, event streams) without loading everything into memory. The caller shouldn't care how data is fetched — page by page, from a cursor, from a stream.

## Full Backend Example: Cursor-Paginated DB Iterator

```python
# iterators.py
from abc import ABC, abstractmethod
from typing import TypeVar, Generic, Iterator, Optional, Callable, List
from dataclasses import dataclass

T = TypeVar("T")

class BatchIterator(Generic[T]):
    """
    Lazily fetches pages from DB or API.
    Caller iterates seamlessly — no paging logic in their code.
    """
    
    def __init__(self,
                 fetch_fn: Callable[[Optional[str], int], tuple[List[T], Optional[str]]],
                 batch_size: int = 100):
        """
        fetch_fn: (cursor, size) → (items, next_cursor)
        Returns (items, None) when no more pages.
        """
        self._fetch = fetch_fn
        self._batch_size = batch_size
        self._buffer: List[T] = []
        self._cursor: Optional[str] = None
        self._exhausted = False
    
    def __iter__(self) -> Iterator[T]:
        return self
    
    def __next__(self) -> T:
        # Refill buffer when empty
        if not self._buffer:
            if self._exhausted:
                raise StopIteration
            
            items, next_cursor = self._fetch(self._cursor, self._batch_size)
            
            if not items:
                raise StopIteration
            
            self._buffer = items
            self._cursor = next_cursor
            
            if next_cursor is None:
                self._exhausted = True
        
        return self._buffer.pop(0)
    
    def in_batches(self, batch_size: int) -> Iterator[List[T]]:
        """Yield items in batches (useful for bulk operations)."""
        batch = []
        for item in self:
            batch.append(item)
            if len(batch) >= batch_size:
                yield batch
                batch = []
        if batch:
            yield batch


# Concrete iterators for common patterns

class OrderIterator(BatchIterator["Order"]):
    """Iterates all orders in DB via cursor pagination."""
    
    def __init__(self, db, filters: dict = None, batch_size: int = 100):
        def fetch(cursor: Optional[str], size: int):
            query = "SELECT * FROM orders"
            params = []
            conditions = []
            
            if cursor:
                conditions.append("id > %s")
                params.append(cursor)
            
            if filters:
                for col, val in filters.items():
                    conditions.append(f"{col} = %s")
                    params.append(val)
            
            if conditions:
                query += " WHERE " + " AND ".join(conditions)
            
            query += f" ORDER BY id LIMIT {size}"
            rows = db.execute(query, params)
            items = [Order.from_row(r) for r in rows]
            next_cursor = str(items[-1].id) if len(items) == size else None
            return items, next_cursor
        
        super().__init__(fetch, batch_size)


class ApiPageIterator(BatchIterator[dict]):
    """Iterates paginated external API responses."""
    
    def __init__(self, http_client, endpoint: str, page_size: int = 50):
        def fetch(cursor: Optional[str], size: int):
            params = {"limit": size}
            if cursor:
                params["cursor"] = cursor
            
            response = http_client.get(endpoint, params=params)
            data = response.json()
            
            items = data["data"]
            next_cursor = data.get("pagination", {}).get("next_cursor")
            return items, next_cursor
        
        super().__init__(fetch, page_size)


# Usage examples

# Process all orders without loading all into memory
def export_all_orders_to_csv(db, output_file: str):
    iterator = OrderIterator(db, filters={"status": "COMPLETED"})
    
    with open(output_file, "w") as f:
        writer = csv.DictWriter(f, fieldnames=["id", "user_id", "total", "created_at"])
        writer.writeheader()
        
        for order in iterator:                    # transparent paging
            writer.writerow(order.to_dict())

# Bulk process in batches
def send_bulk_emails(db):
    iterator = OrderIterator(db, filters={"status": "CONFIRMED"})
    
    for batch in iterator.in_batches(batch_size=50):  # 50 orders at a time
        emails = [build_email(order) for order in batch]
        email_service.send_bulk(emails)

# Export from external API
def sync_products_from_api(http_client):
    iterator = ApiPageIterator(http_client, "/v1/products", page_size=100)
    
    for product in iterator:
        product_repo.upsert(product)
```

## Event Stream Iterator

```python
class KafkaEventIterator:
    """Iterate Kafka messages as a Python iterator."""
    
    def __init__(self, consumer, timeout_ms: int = 5000):
        self._consumer = consumer
        self._timeout = timeout_ms
    
    def __iter__(self): return self
    
    def __next__(self):
        messages = self._consumer.poll(timeout_ms=self._timeout)
        if not messages:
            raise StopIteration
        
        for partition_messages in messages.values():
            for msg in partition_messages:
                return msg.value
        
        raise StopIteration


# Process events one at a time
event_stream = KafkaEventIterator(kafka_consumer)
for event in event_stream:
    handle_event(event)
```

## Internal Collection Iterator

```python
# When you have custom collections that should be iterable
class OrderCollection:
    def __init__(self):
        self._orders: List[Order] = []
    
    def add(self, order: Order): self._orders.append(order)
    
    def __iter__(self) -> Iterator[Order]:
        """Direct iteration — no external iterator class needed."""
        return iter(self._orders)
    
    def pending(self) -> Iterator[Order]:
        """Filtered iterator."""
        return (o for o in self._orders if o.status == OrderStatus.PENDING)
    
    def sorted_by_total(self, descending: bool = False) -> Iterator[Order]:
        return iter(sorted(self._orders, key=lambda o: o.total, reverse=descending))
```

## When NOT to Use Iterator

- **Small collections** — just use a list
- **Random access needed** — iterator is sequential
- **Need to iterate multiple times** — must reset or recreate

## Tradeoffs

|Pro|Con|
|---|---|
|Memory efficient — one item at a time|Sequential only (no random access)|
|Caller agnostic of source|Single-pass (can't rewind without recreating)|
|Lazy evaluation|Error handling mid-stream is tricky|
|Composable with Python's `for` loop||
